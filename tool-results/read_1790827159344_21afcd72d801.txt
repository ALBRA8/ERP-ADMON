    77→// ---------- Carga de certificado PFX/P12 ----------
    78→
    79→/**
    80→ * Carga un certificado PFX/P12 (base64 o Buffer) usando `node-forge`
    81→ * para parsear la estructura PKCS#12, extraer la clave privada y
    82→ * el X.509, y devolverlos en formato PEM. Devuelve la info necesaria
    83→ * para firmar.
    84→ *
    85→ * Si la password es incorrecta, lanza "PKCS#12 password invalid".
    86→ */
    87→export function loadCertificate(
    88→  p12Base64: string | Buffer,
    89→  password: string,
    90→): CertificateInfo {
    91→  const pfxBuffer = Buffer.isBuffer(p12Base64)
    92→    ? p12Base64
    93→    : Buffer.from(p12Base64, "base64");
    94→
    95→  // node-forge trabaja con strings binarios. Convertimos el buffer.
    96→  const p12Der = pfxBuffer.toString("binary");
    97→  const asn1 = forge.asn1.fromDer(p12Der);
    98→  let pfx: any;
    99→  try {
   100→    pfx = forge.pkcs12.pkcs12FromAsn1(asn1, password);
   101→  } catch (e: any) {
   102→    throw new Error(
   103→      "No se pudo abrir el P12. Verifica la contraseña. (" +
   104→        (e?.message ?? "unknown") +
   105→        ")",
   106→    );
   107→  }
   108→
   109→  // Recorrer safeContents -> safeBags buscando certificado y private key.
   110→  let forgeCert: any = null;
   111→  let forgeKey: any = null;
   112→  for (const sc of pfx.safeContents ?? []) {
   113→    for (const bag of sc.safeBags ?? []) {
   114→      if (bag.cert && !forgeCert) forgeCert = bag.cert;
   115→      if (bag.key && !forgeKey) forgeKey = bag.key;
   116→    }
   117→  }
   118→  if (!forgeCert) {
   119→    throw new Error(
   120→      "El P12 no contiene un certificado X.509 en un bag accesible.",
   121→    );
   122→  }
   123→  if (!forgeKey) {
   124→    throw new Error(
   125→      "El P12 no contiene una clave privada accesible. ¿Está protegido con encryptionSafeContainers que no soportamos?",
   126→    );
   127→  }
   128→
   129→  const certPem = forge.pki.certificateToPem(forgeCert);
   130→  const keyPem = forge.pki.privateKeyToPem(forgeKey);
   131→  // DER del certificado (para fingerprint)
   132→  const certDer = forge.asn1.toDer(
   133→    forge.pki.certificateToAsn1(forgeCert),
   134→  ).getBytes();
   135→  const certDerBuf = Buffer.from(certDer, "binary");
   136→
   137→  // Para subject/issuer/serial: usamos crypto.X509Certificate que
   138→  // parsea ASN.1 estándar y devuelve DN legible.
   139→  const x509 = new X509Certificate(certDerBuf);
   140→  const fingerprint = createHash("sha256").update(certDerBuf).digest("hex");
   141→
   142→  return {
   143→    certificatePem: certPem.replace(/\r/g, "").trim() + "\n",
   144→    privateKeyPem: keyPem,
   145→    subject: x509.subject,
   146→    issuer: x509.issuer,
   147→    serialNumber: x509.serialNumber,
   148→    fingerprintSha256: fingerprint,
   149→    validFrom: new Date(x509.validFrom),
   150→    validTo: new Date(x509.validTo),
   151→  };
   152→}
   153→
   154→// ---------- Firma XAdES-EPES envolvente sobre <Invoice> ----------
   155→
   156→/**
   157→ * Aplica firma XAdES-EPES envolvente al elemento raíz <Invoice>
   158→ * con los SignedProperties requeridos por la DIAN Colombia:
   159→ *   - signingTime (fecha de firma ISO)
   160→ *   - signingCertificate (digest SHA-256 del X509)
   161→ *   - dataObjectFormat (descripción del documento)
   162→ *
   163→ * El resultado es el XML original con <ds:Signature> insertado como
   164→ * hijo de <Invoice>, y una referencia con URI vacía al documento entero.
   165→ */
   166→export function signXMLWithXAdES(
   167→  xml: string,
   168→  cert: CertificateInfo,
   169→  cufe: string,
   170→): string {
   171→  const signingTime = new Date().toISOString();
   172→  const certDigestSha256 = createHash("sha256")
   173→    .update(stripPemHeaders(cert.certificatePem))
   174→    .digest("base64");
   175→
   176→  // XAdES Qualifying Properties (envueltas en ds:Object)
   177→  const qualifyingProperties = `\
   178→<xades:QualifyingProperties Target="#xmldsig-${cufe}" xmlns:xades="http://uri.etsi.org/01903/v1.3.2#" xmlns:xades141="http://uri.etsi.org/01903/v1.4.1#">
   179→  <xades:SignedProperties Id="xmldsig-${cufe}-signedprops">
   180→    <xades:SignedSignatureProperties>
   181→      <xades:SigningTime>${signingTime}</xades:SigningTime>
   182→      <xades:SigningCertificate>
   183→        <xades:Cert>
   184→          <xades:CertDigest>
   185→            <ds:DigestMethod Algorithm="http://www.w3.org/2001/04/xmlenc#sha256"/>
   186→            <ds:DigestValue>${certDigestSha256}</ds:DigestValue>
   187→          </xades:CertDigest>
   188→          <xades:IssuerSerial>
   189→            <ds:X509IssuerName>${escapeXml(cert.issuer)}</ds:X509IssuerName>
   190→            <ds:X509SerialNumber>${escapeXml(cert.serialNumber)}</ds:X509SerialNumber>
   191→          </xades:IssuerSerial>
   192→        </xades:Cert>
   193→      </xades:SigningCertificate>
   194→      <xades:SignaturePolicyIdentifier>
   195→        <xades:SignaturePolicyImplied/>
   196→      </xades:SignaturePolicyIdentifier>
   197→    </xades:SignedSignatureProperties>
   198→    <xades:SignedDataObjectProperties>
   199→      <xades:DataObjectFormat ObjectReference="#ref-${cufe}">
   200→        <xades:MimeType>text/xml</xades:MimeType>
   201→        <xades:Encoding>UTF-8</xades:Encoding>
   202→        <xades:Description>Factura Electrónica de Venta UBL 2.1</xades:Description>
   203→      </xades:DataObjectFormat>
   204→    </xades:SignedDataObjectProperties>
   205→  </xades:SignedProperties>
   206→</xades:QualifyingProperties>`;
   207→
   208→  const sig = new SignedXml({
   209→    privateKey: cert.privateKeyPem,
   210→    publicCert: cert.certificatePem,
   211→    signatureAlgorithm: "http://www.w3.org/2001/04/xmldsig-more#rsa-sha256",
   212→    canonicalizationAlgorithm: "http://www.w3.org/TR/2001/REC-xml-c14n-20010315",
   213→    // XAdES Qualifying Properties van dentro de ds:Object
   214→    objects: [
   215→      {
   216→        content: qualifyingProperties,
   217→        attributes: { Id: `xmldsig-${cufe}-object` },
   218→      },
   219→    ],
   220→  });
   221→
   222→  // Referencia 1: documento firmado (URI vacía = documento completo)
   223→  sig.addReference({
   224→    xpath: "//*[local-name()='Invoice']",
   225→    transforms: ["http://www.w3.org/2000/09/xmldsig#enveloped-signature", "http://www.w3.org/TR/2001/REC-xml-c14n-20010315"],
   226→    digestAlgorithm: "http://www.w3.org/2001/04/xmlenc#sha256",
   227→    uri: "",
   228→    isEmptyUri: true,
   229→    id: `ref-${cufe}`,
   230→  });
   231→
   232→  // Referencia 2: SignedProperties (XAdES)
   233→  sig.addReference({
   234→    xpath: "//*[local-name()='SignedProperties']",
   235→    transforms: ["http://www.w3.org/TR/2001/REC-xml-c14n-20010315"],
   236→    digestAlgorithm: "http://www.w3.org/2001/04/xmlenc#sha256",
   237→    uri: `#xmldsig-${cufe}-signedprops`,
   238→    isEmptyUri: false,
   239→    type: "http://uri.etsi.org/01903#SignedProperties",
   240→  });
   241→
   242→  sig.computeSignature(xml, {
   243→    prefix: "ds",
   244→    attrs: { Id: `xmldsig-${cufe}` },
   245→    location: {
   246→      reference: "//*[local-name()='Invoice']",
   247→      action: "prepend",
   248→    },
   249→    existingPrefixes: {
   250→      ds: "http://www.w3.org/2000/09/xmldsig#",
   251→      xades: "http://uri.etsi.org/01903/v1.3.2#",
   252→    },
   253→  });
   254→
   255→  return sig.getSignedXml();
   256→}

   260→/**
   261→ * Construye el envelope SOAP 1.2 para `SendBillSync` y hace POST
   262→ * al WebService de la DIAN. Devuelve trackId / status / errorMessage.
   263→ *
   264→ * Sin token de PAC configurado, lanza error controlado con mensaje claro.
   265→ */
   266→export async function sendBillSyncToDIAN(
   267→  xmlFirmado: string,
   268→  config: DIANConfig,
   269→  fileName = `face_${Date.now()}.xml`,
   270→): Promise<DIANResponse> {
   271→  if (!config.proveedorTecnologicoToken) {
   272→    return {
   273→      isAccepted: false,
   274→      errorMessage:
   275→        "No hay token de proveedor tecnológico configurado. Solicítelo a su PAC.",
   276→    };
   277→  }
   278→
   279→  const contentBase64 = Buffer.from(xmlFirmado, "utf8").toString("base64");
   280→  const soapAction = "http://www.dian.gov.co/servicios/Billing/v1.0/IBilling/SendBillSync";
   281→  const endpoint = DIAN_ENDPOINTS[config.ambiente];
   282→
   283→  const soapEnvelope = `<?xml version="1.0" encoding="UTF-8"?>
   284→<s:Envelope xmlns:s="http://www.w3.org/2003/05/soap-envelope" xmlns:a="http://www.w3.org/2005/08/addressing">
   285→  <s:Header>
   286→    <a:Action s:mustUnderstand="1">${soapAction}</a:Action>
   287→    <a:To s:mustUnderstand="1">${endpoint}</a:To>
   288→    <AuthorizationToken xmlns="http://www.dian.gov.co/servicios/Billing/v1.0">
   289→      <Token>${escapeXml(config.proveedorTecnologicoToken)}</Token>
   290→    </AuthorizationToken>
   291→  </s:Header>
   292→  <s:Body>
   293→    <SendBillSync xmlns="http://www.dian.gov.co/servicios/Billing/v1.0">
   294→      <fileName>${escapeXml(fileName)}</fileName>
   295→      <contentFile>${contentBase64}</contentFile>
   296→      <documentType>01</documentType>
   297→    </SendBillSync>
   298→  </s:Body>
   299→</s:Envelope>`;
   300→
   301→  return postToDian(endpoint, soapEnvelope, soapAction, "SendBillSync");
   302→}

   384→// ---------- POST a DIAN con manejo de errores ----------
   385→
   386→async function postToDian(
   387→  endpoint: string,
   388→  soapEnvelope: string,
   389→  soapAction: string,
   390→  expectedResponseLocalName: string,
   391→): Promise<DIANResponse> {
   392→  try {
   393→    const controller = new AbortController();
   394→    const timeout = setTimeout(() => controller.abort(), 60_000);
   395→
   396→    const response = await fetch(endpoint, {
   397→      method: "POST",
   398→      headers: {
   399→        "Content-Type": `application/soap+xml; charset=utf-8; action="${soapAction}"`,
   400→        "Accept-Encoding": "gzip,deflate",
   401→        "Content-Length": String(Buffer.byteLength(soapEnvelope, "utf8")),
   402→      },
   403→      body: soapEnvelope,
   404→      signal: controller.signal,
   405→    });
   406→
   407→    clearTimeout(timeout);
   408→
   409→    const text = await response.text();
   410→    return parseDianResponse(text, expectedResponseLocalName);
   411→  } catch (e: any) {
   412→    return {
   413→      isAccepted: false,
   414→      errorMessage:
   415→        "Error de red/timeout contactando DIAN: " + (e?.message ?? String(e)),
   416→    };
   417→  }
   418→}
   419→
   420→/**
   421→ * Extrae trackId, status, isValid y errores del XML de respuesta SOAP.
   422→ * La DIAN responde con un objeto que tiene `StatusCode`, `StatusDescription`,
   423→ * y un `ErrorMessage`-array con `statusCode`/`description`/`correspondence`-id.
   424→ */
   425→function parseDianResponse(
   426→  responseXml: string,
   427→  expectedLocalName: string,
   428→): DIANResponse {
   429→  const clean = responseXml.replace(/<\?xml[^>]*\?>/, "").trim();
   430→  // Quick regex-based extraction (avoid xpath dependency).
   431→  const pick = (tag: string) => {
   432→    const re = new RegExp(
   433→      `<[^:>]*:?${tag}[^>]*>([^<]*)</[^:>]*:?${tag}>`,
   434→      "i",
   435→    );
   436→    const m = clean.match(re);
   437→    return m ? m[1].trim() : undefined;
   438→  };
   439→
   440→  const trackId = pick("TrackId");
   441→  const statusCode = pick("StatusCode");
   442→  const statusMessage = pick("StatusMessage") ?? pick("StatusDescription");
   443→  const cufe = pick("UUID");
   444→  const errorMessageDesc = pick("Description");
   445→
   446→  // 00..99 -> DIAN acceptance: 00..09 = aceptado; 99, 100.. = error
   447→  // Convención estándar DIAN: isValid == "true" si StatusCode está entre "00".."09".
   448→  const accepted =
   449→    statusCode !== undefined &&
   450→    /^0[0-9]$/i.test(statusCode) &&
   451→    statusMessage?.toLowerCase().includes("valid");
   452→
   453→  return {
   454→    trackId,
   455→    status: statusCode,
   456→    cufe,
   457→    errorMessage: errorMessageDesc,
   458→    raw: clean,
   459→    isAccepted: Boolean(accepted),
   460→  };
   461→}
   462→
   463→// ---------- Carga de configuración DIAN desde BD ----------
   464→
   465→/**
   466→ * Carga la configuración DIAN activa de una empresa, descifra el
   467→ * certificado P12 y la contraseña, e instancia el CertificateInfo.
   468→ * Devuelve null si no hay configuración activa o no hay certificado.
   469→ */
   470→export async function loadDIANConfigFromDB(
   471→  empresaId: string,
   472→): Promise<{ config: DIANConfig; configRecord: any } | null> {
   473→  const record = await db.configuracionDIAN.findFirst({
   474→    where: {
   475→      empresaId,
   476→      activo: true,
   477→    },
   478→    orderBy: { updatedAt: "desc" },
   479→  });
   480→  if (!record || !record.certificadoP12Enc || !record.certificadoPasswordEnc) {
   481→    return null;
   482→  }
   483→
   484→  try {
   485→    const p12B64 = decrypt(record.certificadoP12Enc);
   486→    const password = decrypt(record.certificadoPasswordEnc);
   487→    const cert = loadCertificate(p12B64, password);
   488→
   489→    return {
   490→      config: {
   491→        ambiente: record.ambiente as "pruebas" | "produccion",
   492→        proveedorTecnologicoNit: record.proveedorTecnologicoNit ?? undefined,
   493→        proveedorTecnologicoToken: record.proveedorTecnologicoTokenEnc
   494→          ? decrypt(record.proveedorTecnologicoTokenEnc)
   495→          : undefined,
   496→        softwareId: record.softwareId ?? undefined,
   497→        softwarePin: record.softwarePin ?? undefined,
   498→        webhookUrl: record.webhookUrl ?? undefined,
   499→        certificate: cert,
   500→      },
   501→      configRecord: record,
   502→    };
   503→  } catch (e: any) {
   504→    console.error(
   505→      "[dian-real] No se pudo cargar el certificado:",
   506→      e?.message ?? e,
   507→    );
   508→    return null;
   509→  }
   510→}

   512→// ---------- Flujo completo: firmar + enviar factura ----------
   513→
   514→/**
   515→ * Toma una factura ya persistida (con xmlUBL), firma con XAdES-EPES,
   516→ * envía a DIAN (SendBillSync), registra en DIANEnvio y actualiza
   517→ * FacturaVenta.estadoDIAN. Si falla, programa reintento.
   518→ */
   519→export async function enviarFacturaADian(
   520→  facturaId: string,
   521→  empresaId: string,
   522→): Promise<{ ok: boolean; envioId?: string; trackId?: string; mensaje?: string }> {
   523→  const factura = await db.facturaVenta.findFirst({
   524→    where: { id: facturaId, empresaId },
   525→  });
   526→  if (!factura) {
   527→    return { ok: false, mensaje: "Factura no encontrada" };
   528→  }
   529→  if (!factura.xmlUBL || !factura.cufe) {
   530→    return { ok: false, mensaje: "La factura no tiene XML UBL ni CUFE generado" };
   531→  }
   532→  if (!["borrador", "procesado", "rechazado"].includes(factura.estadoDIAN)) {
   533→    // ya aceptada o enviada-aceptada, no reenviar salvo explícito
   534→  }
   535→
   536→  const loaded = await loadDIANConfigFromDB(empresaId);
   537→  if (!loaded) {
   538→    return {
   539→      ok: false,
   540→      mensaje:
   541→        "No hay configuración DIAN activa con certificado cargado. Sube tu .p12 en la pestaña Configuración.",
   542→    };
   543→  }
   544→
   545→  const { config, configRecord } = loaded;
   546→
   547→  // 1. Crear registro de envío (pendiente)
   548→  const envio = await db.dIANEnvio.create({
   549→    data: {
   550→      empresaId,
   551→      facturaId: factura.id,
   552→      tipoDocumento: "factura_venta",
   553→      cufe: factura.cufe!,
   554→      estado: "enviando",
   555→      intentos: 1,
   556→      maxIntentos: 5,
   557→      proximoIntento: null,
   558→    },
   559→  });
   560→
   561→  try {
   562→    // 2. Firmar XML
   563→    const xmlFirmado = signXMLWithXAdES(factura.xmlUBL, config.certificate!, factura.cufe!);
   564→    await db.dIANEnvio.update({
   565→      where: { id: envio.id },
   566→      data: { xmlFirmado },
   567→    });
   568→
   569→    // 3. Enviar a DIAN
   570→    const dianResp = await sendBillSyncToDIAN(xmlFirmado, config, `face_${factura.numero}.xml`);
   571→
   572→    if (dianResp.trackId) {
   573→      await db.dIANEnvio.update({
   574→        where: { id: envio.id },
   575→        data: {
   576→          trackId: dianResp.trackId,
   577→          estado: dianResp.isAccepted ? "aceptado" : "pendiente",
   578→          respuestaDian: JSON.stringify({
   579→            statusCode: dianResp.status,
   580→            statusMessage: dianResp.errorMessage,
   581→            cufe: dianResp.cufe,
   582→            raw: dianResp.raw,
   583→          }),
   584→          proximoIntento: dianResp.isAccepted
   585→            ? null
   586→            : new Date(Date.now() + 5 * 60 * 1000),
   587→        },
   588→      });
   589→
   590→      await db.facturaVenta.update({
   591→        where: { id: factura.id },
   592→        data: {
   593→          estadoDIAN: dianResp.isAccepted ? "aceptado" : "procesado",
   594→        },
   595→      });
   596→
   597→      return {
   598→        ok: true,
   599→        envioId: envio.id,
   600→        trackId: dianResp.trackId,
   601→        mensaje: dianResp.isAccepted
   602→          ? "Factura aceptada por DIAN"
   603→          : `Enviada, en proceso (trackId: ${dianResp.trackId})`,
   604→      };
   605→    } else {
   606→      // Error de envío
   607→      await db.dIANEnvio.update({
   608→        where: { id: envio.id },
   609→        data: {
   610→          estado: "error",
   611→          mensajeError: dianResp.errorMessage ?? "Error desconocido",
   612→          respuestaDian: dianResp.raw
   613→            ? JSON.stringify({ raw: dianResp.raw })
   614→            : null,
   615→          proximoIntento: new Date(Date.now() + 5 * 60 * 1000),
   616→        },
   617→      });
   618→
   619→      return {
   620→        ok: false,
   621→        envioId: envio.id,
   622→        mensaje: dianResp.errorMessage ?? "Error de envío DIAN",
   623→      };
   624→    }
   625→  } catch (e: any) {
   626→    const mensaje = e?.message ?? String(e);
   627→    await db.dIANEnvio.update({
   628→      where: { id: envio.id },
   629→      data: {
   630→        estado: "error",
   631→        mensajeError: mensaje,
   632→        proximoIntento: new Date(Date.now() + 5 * 60 * 1000),
   633→      },
   634→    });
   635→    return { ok: false, envioId: envio.id, mensaje };
   636→  }
   637→}
   638→
   639→// ---------- Procesador de cola de reintentos ----------
   640→
   641→/**
   642→ * Procesa todos los DIANEnvio pendientes o con proximoIntento vencido.
   643→ * Aplica backoff exponencial: intento N -> 2^N * 60s.
   644→ */
   645→export async function procesarColaDIAN(
   646→  empresaId: string,
   647→  limite = 50,
   648→): Promise<QueueProcessResult> {
   649→  const now = new Date();
   650→  const pendientes = await db.dIANEnvio.findMany({
   651→    where: {
   652→      empresaId,
   653→      OR: [
   654→        { estado: "pendiente" },
   655→        { estado: "error" },
   656→        { estado: "enviando", proximoIntento: { lte: now } },
   657→      ],
   658→      intentos: { lt: 5 },
   659→    },
   660→    take: limite,
   661→    orderBy: { createdAt: "asc" },
   662→  });
   663→
   664→  const result: QueueProcessResult = {
   665→    processed: 0,
   666→    accepted: 0,
   667→    rejected: 0,
   668→    errors: 0,
   669→    rescheduled: 0,
   670→    details: [],
   671→  };
   672→
   673→  for (const envio of pendientes) {
   674→    if (!envio.facturaId) continue;
   675→    result.processed++;
   676→
   677→    const updated = await db.dIANEnvio.update({
   678→      where: { id: envio.id },
   679→      data: {
   680→        intentos: { increment: 1 },
   681→        estado: "enviando",
   682→      },
   683→    });
   684→
   685→    const res = await enviarFacturaADian(envio.facturaId, empresaId);
   686→
   687→    if (res.ok && res.trackId) {
   688→      result.accepted++;
   689→      result.details.push({
   690→        envioId: envio.id,
   691→        facturaId: envio.facturaId ?? undefined,
   692→        estado: "aceptado",
   693→        mensaje: res.mensaje,
   694→      });
   695→    } else {
   696→      // Si ya llegó a maxIntentos, marcar como rechazado definitivo
   697→      if (updated.intentos >= updated.maxIntentos) {
   698→        await db.dIANEnvio.update({
   699→          where: { id: envio.id },
   700→          data: {
   701→            estado: "rechazado",
   702→            mensajeError: res.mensaje ?? "Máximos reintentos alcanzados",
   703→          },
   704→        });
   705→        result.rejected++;
   706→      } else {
   707→        // Reagendar con backoff exponencial (2^n * 60s)
   708→        const delayMs = Math.pow(2, updated.intentos) * 60 * 1000;
   709→        await db.dIANEnvio.update({
   710→          where: { id: envio.id },
   711→          data: {
   712→            estado: "error",
   713→            mensajeError: res.mensaje,
   714→            proximoIntento: new Date(Date.now() + delayMs),
   715→          },
   716→        });
   717→        result.errors++;
   718→        result.rescheduled++;
   719→      }
   720→      result.details.push({
   721→        envioId: envio.id,
   722→        facturaId: envio.facturaId ?? undefined,
   723→        estado: "error",
   724→        mensaje: res.mensaje,
   725→      });
   726→    }
   727→  }
   728→
   729→  return result;
   730→}

   750→// ---------- Test de conexión DIAN ----------
   751→
   752→/**
   753→ * Prueba la conexión con DIAN en el ambiente configurado.
   754→ * Sin certificado, responde gracefully.
   755→ */
   756→export async function testConnectionDIAN(
   757→  empresaId: string,
   758→): Promise<{ ok: boolean; message: string; ambiente?: string }> {
   759→  const loaded = await loadDIANConfigFromDB(empresaId);
   760→  if (!loaded) {
   761→    return {
   762→      ok: false,
   763→      message:
   764→        "No hay certificado cargado. Sube tu .p12 en la pestaña Configuración.",
   765→    };
   766→  }
   767→
   768→  const { config, configRecord } = loaded;
   769→
   770→  // Validar vigencia del certificado
   771→  const now = new Date();
   772→  if (config.certificate!.validFrom > now) {
   773→    return {
   774→      ok: false,
   775→      message: `El certificado aún no es válido (vigente desde ${config.certificate!.validFrom.toISOString().slice(0, 10)})`,
   776→      ambiente: config.ambiente,
   777→    };
   778→  }
   779→  if (config.certificate!.validTo < now) {
   780→    return {
   781→      ok: false,
   782→      message: `El certificado expiró el ${config.certificate!.validTo.toISOString().slice(0, 10)}. Renueva el P12 con tu CA.`,
   783→      ambiente: config.ambiente,
   784→    };
   785→  }
   786→
   787→  // Si hay token PAC, hacemos ping real a DIAN con un trackId falso
   788→  // (la DIAN responde 404/202, lo que prueba reachability).
   789→  if (config.proveedorTecnologicoToken) {
   790→    const endpoint = DIAN_ENDPOINTS[config.ambiente];
   791→    try {
   792→      const controller = new AbortController();
   793→      const timeout = setTimeout(() => controller.abort(), 10_000);
   794→      const resp = await fetch(endpoint, {
   795→        method: "POST",
   796→        headers: { "Content-Type": "text/xml; charset=utf-8" },
   797→        body: '<?xml version="1.0" encoding="UTF-8"?><s:Envelope xmlns:s="http://www.w3.org/2003/05/soap-envelope"><s:Body/></s:Envelope>',
   798→        signal: controller.signal,
   799→      });
   800→      clearTimeout(timeout);
   801→      // Cualquier respuesta HTTP (incluso 400/500) indica reachability
   802→      return {
   803→        ok: true,
   804→        message: `Conexión exitosa con DIAN ${config.ambiente}. Respuesta HTTP ${resp.status}. Certificado: ${config.certificate!.subject.slice(0, 60)}...`,
   805→        ambiente: config.ambiente,
   806→      };
   807→    } catch (e: any) {
   808→      return {
   809→        ok: false,
   810→        message: `No se pudo conectar con DIAN: ${e?.message ?? e}`,
   811→        ambiente: config.ambiente,
   812→      };
   813→    }
   814→  }
   815→
   816→  // Hay certificado pero no PAC: certificamos la firma localmente
   817→  return {
   818→    ok: true,
   819→    message: `Certificado cargado y vigente (subject: ${config.certificate!.subject.slice(0, 60)}...). Sin token PAC: el envío real requiere credenciales de proveedor tecnológico autorizado.`,
   820→    ambiente: config.ambiente,
   821→  };
   822→}

   824→// ---------- Guardar configuración desde multipart ----------
   825→
   826→/**
   827→ * Recibe los campos del form de configuración DIAN, valida y guarda.
   828→ * Cifra el P12 y la contraseña con AES-256-GCM antes de persistir.
   829→ */
   830→export async function saveDIANConfig(params: {
   831→  empresaId: string;
   832→  ambiente: "pruebas" | "produccion";
   833→  certificadoP12B64?: string; // base64 del .p12
   834→  certificadoPassword?: string;
   835→  proveedorTecnologicoNit?: string;
   836→  proveedorTecnologicoToken?: string;
   837→  softwareId?: string;
   838→  softwarePin?: string;
   839→  webhookUrl?: string;
   840→  activo?: boolean;
   841→}): Promise<{ ok: boolean; config?: any; mensaje: string }> {
   842→  const {
   843→    empresaId,
   844→    ambiente,
   845→    certificadoP12B64,
   846→    certificadoPassword,
   847→    proveedorTecnologicoNit,
   848→    proveedorTecnologicoToken,
   849→    softwareId,
   850→    softwarePin,
   851→    webhookUrl,
   852→    activo = true,
   853→  } = params;
   854→
   855→  let certSubject: string | undefined;
   856→  let certValidFrom: Date | undefined;
   857→  let certValidTo: Date | undefined;
   858→  let p12Enc: string | undefined;
   859→  let passEnc: string | undefined;
   860→
   861→  if (certificadoP12B64 && certificadoPassword) {
   862→    // Carga y valida el certificado antes de guardar
   863→    const info = loadCertificate(certificadoP12B64, certificadoPassword);
   864→    certSubject = info.subject;
   865→    certValidFrom = info.validFrom;
   866→    certValidTo = info.validTo;
   867→    p12Enc = encrypt(certificadoP12B64);
   868→    passEnc = encrypt(certificadoPassword);
   869→  }
   870→
   871→  // Si se activa esta config, desactivar las demás de la empresa
   872→  if (activo) {
   873→    await db.configuracionDIAN.updateMany({
   874→      where: { empresaId, activo: true },
   875→      data: { activo: false },
   876→    });
   877→  }
   878→
   879→  // Upsert por (empresaId, ambiente)
   880→  const existing = await db.configuracionDIAN.findUnique({
   881→    where: {
   882→      empresaId_ambiente: { empresaId, ambiente },
   883→    },
   884→  });
   885→
   886→  const data: any = {
   887→    ambiente,
   888→    proveedorTecnologicoNit,
   889→    proveedorTecnologicoTokenEnc: proveedorTecnologicoToken
   890→      ? encrypt(proveedorTecnologicoToken)
   891→      : existing?.proveedorTecnologicoTokenEnc ?? null,
   892→    softwareId,
   893→    softwarePin,
   894→    webhookUrl,
   895→    activo,
   896→  };
   897→  if (p12Enc) data.certificadoP12Enc = p12Enc;
   898→  if (passEnc) data.certificadoPasswordEnc = passEnc;
   899→  if (certSubject) data.certificadoSubject = certSubject;
   900→  if (certValidFrom) data.certificadoValidFrom = certValidFrom;
   901→  if (certValidTo) data.certificadoValidTo = certValidTo;
   902→
   903→  let record;
   904→  if (existing) {
   905→    record = await db.configuracionDIAN.update({
   906→      where: { id: existing.id },
   907→      data,
   908→    });
   909→  } else {
   910→    record = await db.configuracionDIAN.create({
   911→      data: {
   912→        empresaId,
   913→        ambiente,
   914→        certificadoP12Enc: p12Enc,
   915→        certificadoPasswordEnc: passEnc,
   916→        certificadoSubject: certSubject,
   917→        certificadoValidFrom: certValidFrom,
   918→        certificadoValidTo: certValidTo,
   919→        proveedorTecnologicoNit,
   920→        proveedorTecnologicoTokenEnc: proveedorTecnologicoToken
   921→          ? encrypt(proveedorTecnologicoToken)
   922→          : null,
   923→        softwareId,
   924→        softwarePin,
   925→        webhookUrl,
   926→        activo,
   927→      },
   928→    });
   929→  }
   930→
   931→  return {
   932→    ok: true,
   933→    config: record,
   934→    mensaje: certSubject
   935→      ? `Certificado cargado: ${certSubject.slice(0, 60)}... (válido hasta ${certValidTo?.toISOString().slice(0, 10)})`
   936→      : "Configuración guardada sin certificado nuevo",
   937→  };
   938→}
