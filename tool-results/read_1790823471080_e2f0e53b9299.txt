   190→function formatDate(iso: string | Date) {
   191→  try {
   192→    const d = typeof iso === "string" ? new Date(iso) : iso;
   193→    return d.toLocaleDateString("es-CO", {
   194→      year: "numeric",
   195→      month: "2-digit",
   196→      day: "2-digit",
   197→    });
   198→  } catch {
   199→    return String(iso);
   200→  }
   201→}

   203→function todayISO() {
   204→  const d = new Date();
   205→  const tz = d.getTimezoneOffset() * 60000;
   206→  return new Date(d.getTime() - tz).toISOString().slice(0, 10);
   207→}
   208→
   209→function datePlusDays(days: number): string {
   210→  const d = new Date();
   211→  d.setDate(d.getDate() + days);
   212→  const tz = d.getTimezoneOffset() * 60000;
   213→  return new Date(d.getTime() - tz).toISOString().slice(0, 10);
   214→}

   216→let _keyCounter = 0;
   217→function nextKey() {
   218→  _keyCounter += 1;
   219→  return `it-${Date.now()}-${_keyCounter}`;
   220→}

   222→function round2(n: number) {
   223→  return Math.round((n + Number.EPSILON) * 100) / 100;
   224→}
   225→
   226→function computeLine(l: ItemLine) {
   227→  const gross = round2(l.cantidad * l.precioUnitario);
   228→  const descuento = round2(gross * (l.descuentoPorcentaje / 100));
   229→  const subtotal = round2(gross - descuento);
   230→  const iva = round2(subtotal * (l.tarifaIva / 100));
   231→  const total = round2(subtotal + iva);
   232→  return { gross, descuento, subtotal, iva, total };
   233→}
   234→
   235→function computeTotals(items: ItemLine[]) {
   236→  let subtotalBruto = 0;
   237→  let descuentoTotal = 0;
   238→  let subtotalNeto = 0;
   239→  let totalIva = 0;
   240→  let total = 0;
   241→  for (const l of items) {
   242→    const c = computeLine(l);
   243→    subtotalBruto += c.gross;
   244→    descuentoTotal += c.descuento;
   245→    subtotalNeto += c.subtotal;
   246→    totalIva += c.iva;
   247→    total += c.total;
   248→  }
   249→  return {
   250→    subtotalBruto: round2(subtotalBruto),
   251→    descuentoTotal: round2(descuentoTotal),
   252→    subtotalNeto: round2(subtotalNeto),
   253→    totalIva: round2(totalIva),
   254→    total: round2(total),
   255→  };
   256→}

   406→function ListaFacturas({ onGoNew }: { onGoNew: () => void }) {
   407→  const [facturas, setFacturas] = useState<FacturaItem[]>([]);
   408→  const [loading, setLoading] = useState(true);
   409→  const [search, setSearch] = useState("");
   410→  const [debouncedSearch, setDebouncedSearch] = useState("");
   411→
   412→  const cargar = useCallback(async (q?: string) => {
   413→    setLoading(true);
   414→    try {
   415→      const url = new URL("/api/facturas", window.location.origin);
   416→      url.searchParams.set("pageSize", "100");
   417→      if (q) url.searchParams.set("q", q);
   418→      const r = await fetch(url.toString());
   419→      const d = await r.json();
   420→      if (!r.ok) throw new Error(d.error ?? "Error");
   421→      setFacturas((d.items ?? []) as FacturaItem[]);
   422→    } catch (e: any) {
   423→      toast.error("No se pudo listar facturas", { description: e?.message });
   424→    } finally {
   425→      setLoading(false);
   426→    }
   427→  }, []);
   428→
   429→  useEffect(() => {
   430→    cargar();
   431→  }, [cargar]);
   432→
   433→  // debounce búsqueda
   434→  useEffect(() => {
   435→    const t = setTimeout(() => setDebouncedSearch(search.trim()), 350);
   436→    return () => clearTimeout(t);
   437→  }, [search]);
   438→
   439→  useEffect(() => {
   440→    cargar(debouncedSearch || undefined);
   441→  }, [debouncedSearch, cargar]);
   442→
   443→  const filtered = useMemo(() => {
   444→    return facturas;
   445→  }, [facturas]);
   446→
   447→  return (
   448→    <Card className="border-neutral-200 dark:border-neutral-800">
   449→      <CardHeader className="pb-3">
   450→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   451→          <div>
   452→            <CardTitle className="text-base flex items-center gap-2">
   453→              <ListChecks className="size-4 text-emerald-600 dark:text-emerald-400" />
   454→              Documentos emitidos
   455→            </CardTitle>
   456→            <CardDescription>
   457→              Facturas electrónicas y POS registradas ante la DIAN.
   458→            </CardDescription>
   459→          </div>
   460→          <div className="flex items-center gap-2">
   461→            <div className="relative">
   462→              <Search className="size-4 absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground" />
   463→              <Input
   464→                value={search}
   465→                onChange={(e) => setSearch(e.target.value)}
   466→                placeholder="Buscar número / cliente / NIT"
   467→                className="pl-8 w-full sm:w-72"
   468→              />
   469→            </div>
   470→            <Button
   471→              variant="outline"
   472→              size="icon"
   473→              onClick={() => cargar(debouncedSearch || undefined)}
   474→              title="Refrescar"
   475→            >
   476→              <RefreshCw className="size-4" />
   477→            </Button>
   478→            <Button
   479→              onClick={onGoNew}
   480→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   481→            >
   482→              <Plus className="size-4" />
   483→              <span className="hidden sm:inline">Nueva</span>
   484→            </Button>
   485→          </div>
   486→        </div>
   487→      </CardHeader>
   488→      <CardContent className="pt-0">
   489→        <div className="rounded-md border border-neutral-200 dark:border-neutral-800 overflow-hidden">
   490→          <div className="max-h-[70vh] overflow-auto">
   491→            <Table>
   492→              <TableHeader className="sticky top-0 z-10 bg-neutral-50 dark:bg-neutral-900">
   493→                <TableRow>
   494→                  <TableHead className="min-w-[120px]">Número</TableHead>
   495→                  <TableHead className="min-w-[110px]">Fecha</TableHead>
   496→                  <TableHead className="min-w-[200px]">Cliente</TableHead>
   497→                  <TableHead className="text-right min-w-[120px]">Subtotal</TableHead>
   498→                  <TableHead className="text-right min-w-[110px]">IVA</TableHead>
   499→                  <TableHead className="text-right min-w-[130px]">Total</TableHead>
   500→                  <TableHead className="text-center min-w-[110px]">DIAN</TableHead>
   501→                  <TableHead className="text-center min-w-[120px]">Forma pago</TableHead>
   502→                  <TableHead className="text-right min-w-[120px]">Saldo</TableHead>
   503→                  <TableHead className="text-center min-w-[110px]">Cartera</TableHead>
   504→                </TableRow>
   505→              </TableHeader>
   506→              <TableBody>
   507→                {loading && (

   509→                    <TableCell colSpan={10} className="text-center py-10 text-muted-foreground">
   510→                      Cargando…
   511→                    </TableCell>

   516→                    <TableCell colSpan={10} className="text-center py-10 text-muted-foreground">
   517→                      No hay facturas registradas. Crea la primera.
   518→                    </TableCell>
   519→                  </TableRow>

   522→                  filtered.map((f) => (
   523→                    <TableRow key={f.id} className="hover:bg-neutral-50 dark:hover:bg-neutral-900/60">
   524→                      <TableCell className="font-mono text-xs font-semibold text-emerald-700 dark:text-emerald-400">
   525→                        <span className="inline-flex items-center gap-1">

   527→                          {f.numero}

   546→                        {formaPagoBadge(f.formaPago)}

   549→                        {f.saldoPendiente > 0 ? (
   550→                          <span className="text-rose-600 dark:text-rose-400 font-medium">
   551→                            {formatCurrency(f.saldoPendiente)}
   552→                          </span>
   553→                        ) : (
   554→                          <span className="text-muted-foreground">—</span>
   555→                        )}
   556→                      </TableCell>
   557→                      <TableCell className="text-center">
   558→                        {carteraBadge(f.estadoCartera, f.saldoPendiente)}
   559→                      </TableCell>
   560→                    </TableRow>
   561→                  ))}
   562→              </TableBody>
   563→            </Table>
   564→          </div>
   565→        </div>
   566→      </CardContent>
   567→    </Card>
   568→  );
