   440→function KpiCard({
   441→  label,
   442→  value,
   443→  icon,
   444→  tone,
   445→  sub,
   446→}: {
   447→  label: string;
   448→  value: number;
   449→  icon: React.ReactNode;
   450→  tone: "emerald" | "teal";
   451→  sub?: string;
   452→}) {
   453→  const toneCls =
   454→    tone === "emerald"
   455→      ? "bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300"
   456→      : "bg-teal-50 text-teal-700 dark:bg-teal-950/40 dark:text-teal-300";
   457→  return (
   458→    <Card className="overflow-hidden">
   459→      <CardContent className="p-4">
   460→        <div className="flex items-start justify-between gap-2">
   461→          <div className="min-w-0">
   462→            <p className="truncate text-xs font-medium text-neutral-500 dark:text-neutral-400">
   463→              {label}
   464→            </p>
   465→            <p className="mt-1 truncate text-lg font-semibold text-neutral-900 dark:text-neutral-50">
   466→              {fmtCOP(value)}
   467→            </p>
   468→            {sub ? (
   469→              <p className="mt-0.5 truncate text-xs text-neutral-400 dark:text-neutral-500">
   470→                {sub}
   471→              </p>
   472→            ) : null}
   473→          </div>
   474→          <div className={`rounded-lg p-2 ${toneCls}`}>{icon}</div>
   475→        </div>
   476→      </CardContent>
   477→    </Card>
   478→  );
   479→}

   481→// ============================================================
   482→// Tabla de cartera con aging por edades
   483→// ============================================================
   484→function AgingCartera({
   485→  terceros,
   486→  loading,
   487→  tipo,
   488→  onRecibir,
   489→  onNuevo,
   490→}: {
   491→  terceros: Tercero[];
   492→  loading: boolean;
   493→  tipo: "cliente" | "proveedor";
   494→  onRecibir: (t: Tercero) => void;
   495→  onNuevo: () => void;
   496→}) {
   497→  const [q, setQ] = useState("");
   498→
   499→  // Filtrado: solo terceros con saldoCartera > 0
   500→  const rows = useMemo(() => {
   501→    return terceros
   502→      .filter((t) => (t.saldoCartera ?? 0) > 0)
   503→      .filter((t) => {
   504→        if (!q.trim()) return true;
   505→        const s = q.toLowerCase();
   506→        const nombre = `${t.nombre} ${t.apellido1 ?? ""}`.toLowerCase();
   507→        return (
   508→          nombre.includes(s) ||
   509→          (t.nombreComercial ?? "").toLowerCase().includes(s) ||
   510→          t.numeroDocumento.toLowerCase().includes(s)
   511→        );
   512→      })
   513→      .sort((a, b) => (b.saldoCartera ?? 0) - (a.saldoCartera ?? 0));
   514→  }, [terceros, q]);
   515→
   516→  // Aging simplificado: asumimos todo en rango 0-30
   517→  const totales = useMemo(() => {
   518→    return rows.reduce(
   519→      (acc, t) => {
   520→        const s = t.saldoCartera ?? 0;
   521→        acc.r0_30 += s;
   522→        acc.total += s;
   523→        return acc;
   524→      },
   525→      { r0_30: 0, r31_60: 0, r61_90: 0, r_mas_90: 0, total: 0 },
   526→    );
   527→  }, [rows]);
   528→
   529→  const titulo = tipo === "cliente" ? "Cartera por cobrar (CXC)" : "Cartera por pagar (CXP)";
   530→  const desc =
   531→    tipo === "cliente"
   532→      ? "Clientes con saldo pendiente"
   533→      : "Proveedores con saldo pendiente";
   534→  const accionLabel =
   535→    tipo === "cliente" ? "Recibir pago" : "Pagar a proveedor";
   536→  const nuevoLabel = tipo === "cliente" ? "Nuevo recibo de caja" : "Nuevo egreso";
   537→  const accionIcon =
   538→    tipo === "cliente" ? (
   539→      <ArrowDownToLine className="h-4 w-4" />
   540→    ) : (
   541→      <ArrowUpFromLine className="h-4 w-4" />
   542→    );
   543→
   544→  return (
   545→    <Card>
   546→      <CardHeader className="pb-3">
   547→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   548→          <div>
   549→            <CardTitle className="flex items-center gap-2 text-emerald-700 dark:text-emerald-300">
   550→              {titulo}
   551→            </CardTitle>
   552→            <CardDescription>{desc}</CardDescription>
   553→          </div>
   554→          <Button
   555→            onClick={onNuevo}
   556→            className="bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-700 dark:hover:bg-emerald-600"
   557→            size="sm"
   558→          >
   559→            <Plus className="mr-1.5 h-4 w-4" />
   560→            {nuevoLabel}
   561→          </Button>
   562→        </div>
   563→
   564→        {/* Aviso de datos simplificados */}
   565→        <Alert className="mt-2 border-amber-200 bg-amber-50 text-amber-800 dark:border-amber-900/50 dark:bg-amber-950/30 dark:text-amber-200">
   566→          <AlertCircle className="h-4 w-4" />
   567→          <AlertTitle className="text-xs font-semibold">

   969→      cancelled = true;

   991→      const nombre = `${t.nombre} ${t.apellido1 ?? ""}`.toLowerCase();
   992→      return (
   993→        nombre.includes(s) ||
   994→        (t.nombreComercial ?? "").toLowerCase().includes(s) ||
   995→        t.numeroDocumento.toLowerCase().includes(s)
   996→      );
   997→    });
   998→  }, [terceros, q]);
   999→

  1001→    const n = parseFloat(valorStr.replace(/[^\d.-]/g, ""));
  1002→    return Number.isFinite(n) ? n : 0;
  1003→  }, [valorStr]);
  1004→
  1005→  // Saldo actual del tercero seleccionado (para mostrar en el formulario)
  1006→  const saldoTercero = terceroSel?.saldoCartera ?? 0;
  1007→  const excedeSaldo =
  1008→    terceroSel && valorNum > 0 && valorNum > saldoTercero + 0.01;
  1009→
  1010→  const isValid =
  1011→    !!terceroSel && valorNum > 0 && !excedeSaldo && !saving;
  1012→
  1013→  const tituloDialog =
  1014→    tipo === "recibo" ? "Nuevo recibo de caja" : "Nuevo comprobante de egreso";

  1016→    tipo === "recibo"
  1017→      ? "Recaudo de cartera a un cliente. El saldo del tercero se disminuirá automáticamente."
  1018→      : "Pago a un proveedor o salida de efectivo. El saldo del tercero se disminuirá automáticamente.";

  1023→      : "bg-teal-600 text-white hover:bg-teal-700 dark:bg-teal-700 dark:hover:bg-teal-600";

  1028→      return;
  1029→    }

  1031→      toast.error("El valor debe ser mayor a cero");
  1032→      return;
  1033→    }
  1034→    if (excedeSaldo) {
  1035→      toast.error(
  1036→        `El valor excede el saldo actual del tercero (${fmtCOP(saldoTercero)})`,
  1037→      );
  1038→      return;
  1039→    }
  1040→    setSaving(true);
  1041→    try {
  1042→      const body = {
  1043→        tipo,
  1044→        terceroId: terceroSel.id,
  1045→        nombreTercero:
  1046→          `${terceroSel.nombre}${terceroSel.apellido1 ? " " + terceroSel.apellido1 : ""}`.trim(),
  1047→        valorTotal: valorNum,
  1048→        concepto: concepto.trim() || undefined,
  1049→        formaPago,
  1050→        observaciones: observaciones.trim() || undefined,
  1051→      };
  1052→      const r = await fetch("/api/cartera", {
  1053→        method: "POST",

  1055→        body: JSON.stringify(body),
  1056→      });
  1057→      const d = await r.json();
  1058→      if (!r.ok) {
  1059→        throw new Error(d?.error || "Error al guardar el comprobante");

  1062→        `${tipo === "recibo" ? "Recibo de caja" : "Comprobante de egreso"} creado: ${d.numero}`,

  1068→    } finally {
  1069→      setSaving(false);
  1070→    }
  1071→  };
  1072→
  1073→  return (
  1074→    <Dialog open={open} onOpenChange={onOpenChange}>
  1075→      <DialogContent className="sm:max-w-2xl max-h-[92vh] overflow-y-auto">
  1076→        <DialogHeader>
  1077→          <DialogTitle className="flex items-center gap-2 text-emerald-700 dark:text-emerald-300">
  1078→            {tipo === "recibo" ? (
  1079→              <ReceiptText className="h-5 w-5" />
  1080→            ) : (
  1081→              <FileMinus className="h-5 w-5" />
  1082→            )}
  1083→            {tituloDialog}
  1084→          </DialogTitle>
  1085→          <DialogDescription>{descDialog}</DialogDescription>
