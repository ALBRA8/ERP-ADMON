   104→// ---------- ARL: porcentaje según nivel de riesgo ----------
   105→// Tabla oficial Decreto 1607 de 2013 (5 clases de riesgo)
   106→const ARL_PORCENTAJES: Record<TipoRiesgoARL, number> = {
   107→  1: 0.522,
   108→  2: 1.044,
   109→  3: 2.436,
   110→  4: 4.35,
   111→  5: 6.96,
   112→};
   113→
   114→export function arlNivelFromString(arl?: string | null): TipoRiesgoARL {
   115→  if (!arl) return 1;
   116→  const m = String(arl).match(/([1-5])/);
   117→  if (m) return Number(m[1]) as TipoRiesgoARL;
   118→  // Si es nombre textual de riesgo (bajo, medio, alto...)
   119→  const s = String(arl).toLowerCase();
   120→  if (s.includes("minimo") || s.includes("bajo") || s.includes("nivel i"))
   121→    return 1;
   122→  if (s.includes("ligero") || s.includes("nivel ii")) return 2;
   123→  if (s.includes("medio") || s.includes("nivel iii")) return 3;
   124→  if (s.includes("alto") || s.includes("nivel iv")) return 4;
   125→  if (s.includes("maximo") || s.includes("muy alto") || s.includes("nivel v"))
   126→    return 5;
   127→  return 1;
   128→}

   130→// ============================================================
   131→// 1. Salario por días (regla de 30 días)
   132→// ============================================================
   133→export function calcularSalarioPorDias(
   134→  salarioMensual: number,
   135→  diasLaborados: number,
   136→  diasMes = 30,
   137→): number {
   138→  if (!Number.isFinite(salarioMensual) || salarioMensual < 0) return 0;
   139→  if (!Number.isFinite(diasLaborados) || diasLaborados <= 0) return 0;
   140→  return round((salarioMensual / diasMes) * diasLaborados, 2);
   141→}
   142→
   143→// ============================================================
   144→// 2. Auxilio de transporte (sólo si salario <= 2 SMMLV)
   145→// ============================================================
   146→export function calcularAuxilioTransporte(
   147→  salario: number,
   148→  smmlv: number = SMMLV_2026,
   149→  auxTransporte: number = AUX_TRANSPORTE_2026,
   150→  diasLaborados: number,
   151→  diasMes = 30,
   152→): number {
   153→  if (!Number.isFinite(salario) || salario <= 0) return 0;
   154→  if (salario > 2 * smmlv) return 0;
   155→  if (diasLaborados <= 0) return 0;
   156→  return round((auxTransporte / diasMes) * diasLaborados, 2);
   157→}
   158→
   159→// ============================================================
   160→// 3. Horas extras y recargos
   161→//    - Extra diurna: +25% sobre valor hora ordinaria
   162→//    - Extra nocturna: +75% sobre valor hora ordinaria
   163→//    - Recargo nocturno: +35% sobre valor hora ordinaria
   164→//    - Recargo dominical: +75% sobre valor hora ordinaria
   165→// ============================================================
   166→export function valorHoraOrdinaria(
   167→  salarioMensual: number,
   168→  diasMes = 30,
   169→  horasDia = 8,
   170→): number {
   171→  const horasMes = diasMes * horasDia;
   172→  if (horasMes <= 0) return 0;
   173→  return round(salarioMensual / horasMes, 2);
   174→}
   175→
   176→export function calcularHorasExtras(
   177→  cantidad: number,
   178→  valorHora: number,
   179→  recargo: number, // porcentaje adicional (25, 75, 35)
   180→): number {
   181→  if (!Number.isFinite(cantidad) || cantidad <= 0) return 0;
   182→  return round(cantidad * valorHora * (1 + recargo / 100), 2);
   183→}
   184→
   185→// ============================================================
   186→// 4. IBC (Ingreso Base de Cotización)
   187→//    IBC = devengado constitutivo de salario (excluye auxilio transporte)
   188→// ============================================================
   189→export function calcularIBC(devengadoConstitutivo: number): number {
   190→  if (!Number.isFinite(devengadoConstitutivo) || devengadoConstitutivo < 0)
   191→    return 0;
   192→  // Tope máximo: 25 SMMLV (Ley 1607 de 2013)
   193→  const tope = 25 * SMMLV_2026;
   194→  return round(Math.min(devengadoConstitutivo, tope), 2);
   195→}
   196→
   197→// ============================================================
   198→// 5. Deducción EPS del trabajador: 4% del IBC
   199→// ============================================================
   200→export function calcularDeduccionEPS(
   201→  epsId: string | null | undefined,
   202→  ibc: number,
   203→): number {
   204→  if (!epsId) return 0; // sin EPS registrada, no se deduce
   205→  if (!Number.isFinite(ibc) || ibc <= 0) return 0;
   206→  return round(ibc * 0.04, 2);
   207→}
   208→
   209→// ============================================================
   210→// 6. Deducción Pensión del trabajador: 4% del IBC
   211→// ============================================================
   212→export function calcularDeduccionPension(
   213→  afpId: string | null | undefined,
   214→  ibc: number,
   215→): number {
   216→  if (!afpId) return 0;
   217→  if (!Number.isFinite(ibc) || ibc <= 0) return 0;
   218→  return round(ibc * 0.04, 2);
   219→}
   220→
   221→// ============================================================
   222→// 7. Aportes patronales
   223→//    - ARL: 0.522% a 6.96% según nivel de riesgo (sobre IBC)
   224→//    - ICBF: 3% sobre IBC hasta 10 SMMLV
   225→//    - Caja de compensación: 4% sobre IBC
   226→//    - SENA: 2% sobre IBC (exonerado si < 10 SMMLV en some regímenes)
   227→//    - Pensión patronal: 12% sobre 10 SMMLV (piso)
   228→//    (Simplificado: aplicamos SENA e ICBF sobre el IBC completo)
   229→// ============================================================
   230→export interface AportesPatronalesResult {
   231→  arl: number;
   232→  icbf: number;
   233→  caja: number;
   234→  sena: number;
   235→  pensionPatronal: number;
   236→  total: number;
   237→}
   238→
   239→export function calcularAportesPatronales(
   240→  ibc: number,
   241→  arlRiesgo: TipoRiesgoARL = 1,
   242→): AportesPatronalesResult {
   243→  const base = Number.isFinite(ibc) && ibc > 0 ? ibc : 0;
   244→  const tope10SMMLV = 10 * SMMLV_2026;
   245→  // ARL sobre el IBC
   246→  const arlTasa = ARL_PORCENTAJES[arlRiesgo] ?? ARL_PORCENTAJES[1];
   247→  const arl = round(base * (arlTasa / 100), 2);
   248→  // ICBF 3% sobre IBC hasta 10 SMMLV
   249→  const baseIcbf = Math.min(base, tope10SMMLV);
   250→  const icbf = round(baseIcbf * 0.03, 2);
   251→  // Caja 4% sobre IBC
   252→  const caja = round(base * 0.04, 2);
   253→  // SENA 2% sobre IBC (simplificación: aplicamos siempre)
   254→  const sena = round(base * 0.02, 2);
   255→  // Pensión patronal 12% sobre el IBC hasta 10 SMMLV (piso ARL)
   256→  const pensionPatronal = round(Math.min(base, tope10SMMLV) * 0.12, 2);
   257→  const total = round(arl + icbf + caja + sena + pensionPatronal, 2);
   258→  return { arl, icbf, caja, sena, pensionPatronal, total };
   259→}
   260→
   261→// ============================================================
   262→// 8. Provisiones (prestaciones sociales)
   263→//    - Cesantías: 8.33% sobre devengado constitutivo
   264→//    - Intereses de cesantías: 1% anual sobre cesantías (12%)
   265→//    - Prima: 8.33% sobre devengado constitutivo
   266→//    - Vacaciones: 4.17% sobre el salario (no incluye extras)
   267→// ============================================================
   268→export interface ProvisionesResult {
   269→  cesantias: number;
   270→  interesesCesantias: number;
   271→  prima: number;
   272→  vacaciones: number;
   273→  total: number;
   274→}
   275→
   276→export function calcularProvisiones(
   277→  devengadoConstitutivo: number,
   278→  diasLaborados: number,
   279→  diasMes = 30,
   280→): ProvisionesResult {
   281→  const base = Number.isFinite(devengadoConstitutivo)
   282→    ? devengadoConstitutivo
   283→    : 0;
   284→  // Base proporcional al período (ej: si eran 30 días, base completa; si 15, mitad)
   285→  const factor = diasMes > 0 ? diasLaborados / diasMes : 0;
   286→  // Cesantías: 8.33% (1 año de salario por año trabajado = 8.33% mensual acumulado)
   287→  const cesantias = round(base * 0.0833 * factor, 2);
   288→  // Intereses cesantías: 1% (12% anual sobre las cesantías provisionadas)
   289→  const interesesCesantias = round(base * 0.01 * factor, 2);
   290→  // Prima: 8.33% (un salario por año = 8.33% mensual)
   291→  const prima = round(base * 0.0833 * factor, 2);
   292→  // Vacaciones: 4.17% (medio salario por año = 4.17% mensual)
   293→  // Las vacaciones se calculan sobre el salario base, sin extras ni auxilios.
   294→  // Aquí simplificamos usando la misma base.
   295→  const vacaciones = round(base * 0.0417 * factor, 2);
   296→  const total = round(cesantias + interesesCesantias + prima + vacaciones, 2);
   297→  return { cesantias, interesesCesantias, prima, vacaciones, total };
   298→}
   299→
   300→// ============================================================
   301→// 9. Retención en la fuente por salarios (DIAN 2026)
   302→//    Aplica la tabla de retención con descuento por dependientes
   303→//    (10% de descuento de la base imponible por cada dependiente,
   304→//     hasta máximo 10 dependientes — Estatuto Tributario art. 383).
   305→// ============================================================
   306→export interface RetencionFuenteTabla {
   307→  desde: number;
   308→  hasta: number;
   309→  porcentaje: number;
   310→  baseUVT: number;
   311→}
   312→
   313→export function calcularRetencionSalarios(
   314→  devengadoMenosDeduccionesSalud: number,
   315→  dependientes: number,
   316→  tabla: RetencionFuenteTabla[],
   317→): number {
   318→  if (!Number.isFinite(devengadoMenosDeduccionesSalud) || devengadoMenosDeduccionesSalud <= 0)
   319→    return 0;
   320→  // Deducción por dependientes: 10% por cada uno, máximo 10 (100%)
   321→  const pctDep = Math.min(dependientes, 10) * 0.1;
   322→  const base = devengadoMenosDeduccionesSalud * (1 - pctDep);
   323→  // Buscar rango en la tabla
   324→  const fila = tabla.find((r) => base >= r.desde && base <= r.hasta);
   325→  if (!fila) return 0;
   326→  // Base de retención = base imponible - desde (UVT del tramo)
   327→  const baseRetencion = Math.max(0, base - fila.desde);
   328→  return round(baseRetencion * (fila.porcentaje / 100), 2);
   329→}

   331→// ============================================================
   332→// 10. Cálculo completo de un empleado
   333→//     Genera la liquidación de devengos, deducciones,
   334→//     aportes patronales y provisiones en una sola pasada.
   335→// ============================================================
   336→export function liquidarEmpleado(
   337→  empleado: EmpleadoNominaInput,
   338→  tablaRetencion: RetencionFuenteTabla[] = [],
   339→): ResultadoEmpleadoNomina {
   340→  const salarioPorDias = calcularSalarioPorDias(
   341→    empleado.salario,
   342→    empleado.diasLaborados,
   343→  );
   344→  const auxTransporte = calcularAuxilioTransporte(
   345→    empleado.salario,
   346→    SMMLV_2026,
   347→    AUX_TRANSPORTE_2026,
   348→    empleado.diasLaborados,
   349→  );
   350→
   351→  // Horas extras y recargos
   352→  const vHora = valorHoraOrdinaria(empleado.salario);
   353→  const heDiurnas = empleado.horasExtrasDiurnas
   354→    ? calcularHorasExtras(empleado.horasExtrasDiurnas, vHora, 25)
   355→    : 0;
   356→  const heNocturnas = empleado.horasExtrasNocturnas
   357→    ? calcularHorasExtras(empleado.horasExtrasNocturnas, vHora, 75)
   358→    : 0;
   359→  const recNocturno = empleado.recargosNocturnos
   360→    ? calcularHorasExtras(empleado.recargosNocturnos, vHora, 35)
   361→    : 0;
   362→  const recDominical = empleado.recargosDominicales
   363→    ? calcularHorasExtras(empleado.recargosDominicales, vHora, 75)
   364→    : 0;
   365→
   366→  // Devengos
   367→  const devengos: ConceptoLinea[] = [
   368→    { codigo: "SALARIO_BASICO", nombre: "Salario básico", valor: salarioPorDias },
   369→  ];
   370→  if (auxTransporte > 0)
   371→    devengos.push({
   372→      codigo: "AUX_TRANSPORTE",
   373→      nombre: "Auxilio de transporte",
   374→      valor: auxTransporte,
   375→    });
   376→  if (heDiurnas > 0)
   377→    devengos.push({
   378→      codigo: "HORA_EXTRA_DIURNA",
   379→      nombre: "Horas extras diurnas (25%)",
   380→      valor: heDiurnas,
   381→    });
   382→  if (heNocturnas > 0)
   383→    devengos.push({
   384→      codigo: "HORA_EXTRA_NOCTURNA",
   385→      nombre: "Horas extras nocturnas (75%)",
   386→      valor: heNocturnas,
   387→    });
   388→  if (recNocturno > 0)
   389→    devengos.push({
   390→      codigo: "RECARGO_NOCTURNO",
   391→      nombre: "Recargo nocturno (35%)",
   392→      valor: recNocturno,
   393→    });
   394→  if (recDominical > 0)
   395→    devengos.push({
   396→      codigo: "RECARGO_DOMINICAL",
   397→      nombre: "Recargo dominical (75%)",
   398→      valor: recDominical,
   399→    });
   400→
   401→  // IBC = devengado constitutivo de salario (sin auxilio de transporte)
   402→  const devengadoConstitutivo = round(
   403→    salarioPorDias + heDiurnas + heNocturnas + recNocturno + recDominical,
   404→    2,
   405→  );
   406→  const ibc = calcularIBC(devengadoConstitutivo);
   407→
   408→  // Deducciones
   409→  const deduccionEPS = calcularDeduccionEPS(empleado.epsId, ibc);
   410→  const deduccionPension = calcularDeduccionPension(empleado.afpId, ibc);
   411→
   412→  // Devengado total (incluye auxilio transporte)
   413→  const totalDevengado = round(
   414→    devengos.reduce((a, d) => a + d.valor, 0),
   415→    2,
   416→  );
   417→
   418→  // Retención en la fuente: base = total devengado - EPS - Pensión
   419→  const baseRetencion = round(totalDevengado - deduccionEPS - deduccionPension, 2);
   420→  const retencion = calcularRetencionSalarios(
   421→    baseRetencion,
   422→    empleado.dependientes ?? 0,
   423→    tablaRetencion,
   424→  );
   425→
   426→  const deducciones: ConceptoLinea[] = [];
   427→  if (deduccionEPS > 0)
   428→    deducciones.push({
   429→      codigo: "EPS_TRABAJADOR",
   430→      nombre: `EPS (${empleado.epsId ?? "EPS"}) 4%`,
   431→      valor: deduccionEPS,
   432→    });
   433→  if (deduccionPension > 0)
   434→    deducciones.push({
   435→      codigo: "PENSION_TRABAJADOR",
   436→      nombre: `Pensión (${empleado.afpId ?? "AFP"}) 4%`,
   437→      valor: deduccionPension,
   438→    });
   439→  if (retencion > 0)
   440→    deducciones.push({
   441→      codigo: "RETENCION_FUENTE",
   442→      nombre: "Retención en la fuente salarios",
   443→      valor: retencion,
   444→    });
   445→
   446→  const totalDeducciones = round(
   447→    deducciones.reduce((a, d) => a + d.valor, 0),
   448→    2,
   449→  );
   450→
   451→  // Aportes patronales
   452→  const arlRiesgo = arlNivelFromString(empleado.arl);
   453→  const aportes = calcularAportesPatronales(ibc, arlRiesgo);
   454→  const aportesPatronales: ConceptoLinea[] = [
   455→    { codigo: "ARL_PATRONAL", nombre: `ARL riesgo ${arlRiesgo}`, valor: aportes.arl },
   456→    { codigo: "ICBF_PATRONAL", nombre: "ICBF patronal 3%", valor: aportes.icbf },
   457→    { codigo: "CAJA_COMPENSACION", nombre: "Caja de compensación 4%", valor: aportes.caja },
   458→    { codigo: "SENA_PATRONAL", nombre: "SENA patronal 2%", valor: aportes.sena },
   459→    { codigo: "PENSION_PATRONAL", nombre: "Pensión patronal 12%", valor: aportes.pensionPatronal },
   460→  ];
   461→  const totalAportesPatronales = round(aportes.total, 2);
   462→
   463→  // Provisiones
   464→  const prov = calcularProvisiones(devengadoConstitutivo, empleado.diasLaborados);
   465→  const provisiones: ConceptoLinea[] = [
   466→    { codigo: "CESANTIAS", nombre: "Cesantías 8.33%", valor: prov.cesantias },
   467→    { codigo: "INTERESES_CESANTIAS", nombre: "Intereses cesantías 1%", valor: prov.interesesCesantias },
   468→    { codigo: "PRIMA", nombre: "Prima de servicios 8.33%", valor: prov.prima },
   469→    { codigo: "VACACIONES", nombre: "Vacaciones 4.17%", valor: prov.vacaciones },
   470→  ];
   471→  const totalProvisiones = round(prov.total, 2);
   472→
   473→  const netoAPagar = round(totalDevengado - totalDeducciones, 2);
   474→
   475→  return {
   476→    empleado,
   477→    salarioBasico: salarioPorDias,
   478→    diasLaborados: empleado.diasLaborados,
   479→    ibc,
   480→    devengos,
   481→    deducciones,
   482→    aportesPatronales,
   483→    provisiones,
   484→    totalDevengado,
   485→    totalDeducciones,
   486→    totalAportesPatronales,
   487→    totalProvisiones,
   488→    netoAPagar,
   489→  };
   490→}

   492→// ============================================================
   493→// 11. Cálculo del CUNE (Código Único de Nómina Electrónica)
   494→//     Algoritmo oficial DIAN (Resolución 56 de 2018, anexo técnico):
   495→//     CUNE = SHA-384(numNom + fecGen + horGen + valTotal + codDoc +
   496→//                    nitEmpleador + tipNotDev + nitReceptor + ambiente + 'N')
   497→// ============================================================
   498→export interface CUNEInput {
   499→  numNom: string; // número de nómina (NOM-2026-0001)
   500→  fecGen: string; // YYYY-MM-DD
   501→  horGen: string; // HH:MM:SS
   502→  valTotal: number; // neto a pagar
   503→  codDoc: string; // código de documento (Nómina = "102" | "103" ajuste)
   504→  nitEmpleador: string; // NIT del empleador
   505→  tipNotDev: string; // tipo de nota devolutiva ("1" | "0")
   506→  nitReceptor: string; // NIT/CC del empleado
   507→  ambiente: "pruebas" | "produccion";
   508→}
   509→
   510→export function calcularCUNE(input: CUNEInput): string {
   511→  const partes = [
   512→    input.numNom,
   513→    input.fecGen,
   514→    input.horGen,
   515→    String(round(input.valTotal, 2)),
   516→    input.codDoc,
   517→    input.nitEmpleador,
   518→    input.tipNotDev,
   519→    input.nitReceptor,
   520→    input.ambiente === "produccion" ? "1" : "0",
   521→    "N", // sufijo fijo de Nómina electrónica
   522→  ];
   523→  return createHash("sha384")
   524→    .update(partes.join(""), "utf8")
   525→    .digest("hex");
   526→}

   528→// ============================================================
   529→// 12. Generación XML UBL 2.1 para Nómina Electrónica DIAN
   530→//     Estructura Documento de Pago / Nómina Electrónica.
   531→//     Namespaces:
   532→//       xmlns        = urn:oasis:names:specification:ubl:schema:xsd:NominaDocument-2
   533→//       xmlns:cac    = CommonAggregateComponents-2
   534→//       xmlns:cbc    = CommonBasicComponents-2
   535→//       xmlns:nomina = urn:oasis:names:specification:ubl:schema:xsd:Nomina-2
   536→// ============================================================
   537→function escXML(s: string): string {
   538→  return String(s ?? "")
   539→    .replace(/&/g, "&amp;")
   540→    .replace(/</g, "&lt;")
   541→    .replace(/>/g, "&gt;")
   542→    .replace(/"/g, "&quot;")
   543→    .replace(/'/g, "&apos;");
   544→}
   545→
   546→const fmtN = (n: number) =>
   547→  round(Number.isFinite(n) ? n : 0, 2)
   548→    .toFixed(2)
   549→    .replace(/(\.\d*?)0+$/, "$1")
   550→    .replace(/\.$/, "");
   551→
   552→export interface NominaXMLInput {
   553→  empresa: EmpresaNominaInput;
   554→  empleado: EmpleadoNominaInput;
   555→  resultado: ResultadoEmpleadoNomina;
   556→  nomina: NominaPayload;
   557→  cune: string;
   558→  fechaGen: string; // YYYY-MM-DD
   559→  horaGen: string; // HH:MM:SS
   560→}
   561→
   562→export function generarXMLNomina(input: NominaXMLInput): string {
   563→  const {
   564→    empresa,
   565→    empleado,
   566→    resultado,
   567→    nomina,
   568→    cune,
   569→    fechaGen,
   570→    horaGen,
   571→  } = input;
   572→
   573→  // Mapear tipo de documento del empleado a código DIAN
   574→  const mapTipoDoc: Record<string, string> = {
   575→    CC: "13",
   576→    NIT: "31",
   577→    CE: "22",
   578→    TI: "12",
   579→    PT: "41",
   580→  };
   581→  const codTipoEmp = mapTipoDoc[empleado.tipoDocumento] ?? "13";
   582→
   583→  // Devengos como líneas XML
   584→  const devengosXML = resultado.devengos
   585→    .map(
   586→      (d) => `      <nomina:Devengado>
   587→        <nomina:CodigoConcepto>${escXML(d.codigo)}</nomina:CodigoConcepto>
   588→        <nomina:NombreConcepto>${escXML(d.nombre)}</nomina:NombreConcepto>
   589→        <cbc:Amount currencyID="COP">${fmtN(d.valor)}</cbc:Amount>
   590→      </nomina:Devengado>`,
   591→    )
   592→    .join("\n");
   593→
   594→  const deduccionesXML = resultado.deducciones
   595→    .map(
   596→      (d) => `      <nomina:Deduccion>
   597→        <nomina:CodigoConcepto>${escXML(d.codigo)}</nomina:CodigoConcepto>
   598→        <nomina:NombreConcepto>${escXML(d.nombre)}</nomina:NombreConcepto>
   599→        <cbc:Amount currencyID="COP">${fmtN(d.valor)}</cbc:Amount>
   600→      </nomina:Deduccion>`,
   601→    )
   602→    .join("\n");
   603→
   604→  // Tipo de nómina: código DIAN (1=prestacional, 2=liquidación definitivo)
   605→  const tipoNom = nomina.tipo === "liquidacion_final" ? "2" : "1";
   606→
   607→  return `<?xml version="1.0" encoding="UTF-8"?>
   608→<NominaDocument
   609→  xmlns="urn:oasis:names:specification:ubl:schema:xsd:NominaDocument-2"
   610→  xmlns:cac="urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2"
   611→  xmlns:cbc="urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2"
   612→  xmlns:nomina="urn:oasis:names:specification:ubl:schema:xsd:Nomina-2">
   613→  <cbc:CustomizationID>17</cbc:CustomizationID>
   614→  <cbc:ProfileID>2</cbc:ProfileID>
   615→  <cbc:ID>${escXML(nomina.numero)}</cbc:ID>
   616→  <cbc:UUID schemeID="CUNE">${escXML(cune)}</cbc:UUID>
   617→  <cbc:IssueDate>${escXML(fechaGen)}</cbc:IssueDate>
   618→  <cbc:IssueTime>${escXML(horaGen)}</cbc:IssueTime>
   619→  <nomina:TipoNomina>${tipoNom}</nomina:TipoNomina>
   620→  <cbc:DocumentCurrencyCode>COP</cbc:DocumentCurrencyCode>
   621→  <cac:Period>
   622→    <cbc:StartDate>${escXML(new Date(nomina.periodoInicio).toISOString().slice(0, 10))}</cbc:StartDate>
   623→    <cbc:EndDate>${escXML(new Date(nomina.periodoFin).toISOString().slice(0, 10))}</cbc:EndDate>
   624→  </cac:Period>
   625→  <cac:AccountingSupplierParty>
   626→    <cac:Party>
   627→      <cac:PartyIdentification>
   628→        <cbc:ID schemeID="4">${escXML(empresa.nit)}</cbc:ID>
   629→      </cac:PartyIdentification>
   630→      <cac:PartyName>
   631→        <cbc:Name>${escXML(empresa.nombreComercial ?? empresa.razonSocial)}</cbc:Name>
   632→      </cac:PartyName>
   633→      <cac:PostalAddress>
   634→        <cbc:ID>${escXML(empresa.direccion ?? "N/A")}</cbc:ID>
   635→        <cbc:CityName>${escXML(empresa.ciudad ?? "Bogotá")}</cbc:CityName>
   636→      </cac:PostalAddress>
   637→      <cac:PartyTaxScheme>
   638→        <cbc:RegistrationName>${escXML(empresa.razonSocial)}</cbc:RegistrationName>
   639→        <cbc:CompanyID>${escXML(empresa.nit)}</cbc:CompanyID>
   640→      </cac:PartyTaxScheme>
   641→    </cac:Party>
   642→  </cac:AccountingSupplierParty>
   643→  <cac:AccountingCustomerParty>
   644→    <cac:Party>
   645→      <cac:PartyIdentification>
   646→        <cbc:ID schemeID="${escXML(codTipoEmp)}">${escXML(empleado.numeroDocumento)}</cbc:ID>
   647→      </cac:PartyIdentification>
   648→      <cac:PartyName>
   649→        <cbc:Name>${escXML(empleado.nombre)}</cbc:Name>
   650→      </cac:PartyName>
   651→      <cac:PostalAddress>
   652→        <cbc:ID>${escXML(empleado.direccion ?? "N/A")}</cbc:ID>
   653→        <cbc:CityName>${escXML(empleado.ciudad ?? "Bogotá")}</cbc:CityName>
   654→      </cac:PostalAddress>
   655→      <cac:PartyTaxScheme>
   656→        <cbc:RegistrationName>${escXML(empleado.nombre)}</cbc:RegistrationName>
   657→        <cbc:CompanyID>${escXML(empleado.numeroDocumento)}</cbc:CompanyID>
   658→      </cac:PartyTaxScheme>
   659→    </cac:Party>
   660→  </cac:AccountingCustomerParty>
   661→  <cac:PaymentMilestone>
   662→    <cbc:PaymentID>${escXML(nomina.numero)}</cbc:PaymentID>
   663→    <cbc:PaymentDate>${escXML(new Date(nomina.fechaPago).toISOString().slice(0, 10))}</cbc:PaymentDate>
   664→    <cac:PaymentAmount>
   665→      <cbc:Amount currencyID="COP">${fmtN(resultado.netoAPagar)}</cbc:Amount>
   666→    </cac:PaymentAmount>
   667→    <nomina:DevengadosTotal currencyID="COP">${fmtN(resultado.totalDevengado)}</nomina:DevengadosTotal>
   668→    <nomina:DeduccionesTotal currencyID="COP">${fmtN(resultado.totalDeducciones)}</nomina:DeduccionesTotal>
   669→    <nomina:ComprobanteTotal currencyID="COP">${fmtN(resultado.netoAPagar)}</nomina:ComprobanteTotal>
   670→${devengosXML}
   671→${deduccionesXML}
   672→  </cac:PaymentMilestone>
   673→  <cac:LegalMonetaryTotal>
   674→    <cbc:LineExtensionAmount currencyID="COP">${fmtN(resultado.totalDevengado)}</cbc:LineExtensionAmount>
   675→    <cbc:TaxExclusiveAmount currencyID="COP">${fmtN(resultado.totalDevengado - resultado.totalDeducciones)}</cbc:TaxExclusiveAmount>
   676→    <cbc:PayableAmount currencyID="COP">${fmtN(resultado.netoAPagar)}</cbc:PayableAmount>
   677→  </cac:LegalMonetaryTotal>
   678→</NominaDocument>`;
   679→}

   682→// 13. Helper: generar número consecutivo de nómina

   684→export function generarNumeroNomina(consecutivo: number, year = new Date().getFullYear()): string {
   685→  return `NOM-${year}-${String(consecutivo).padStart(6, "0")}`;
   686→}
