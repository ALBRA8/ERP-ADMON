    22→const fmtCOP = (n: number): string => {
    23→  const v = Number.isFinite(n) ? n : 0;
    24→  return new Intl.NumberFormat("es-CO", {
    25→    style: "currency",
    26→    currency: "COP",
    27→    minimumFractionDigits: 0,
    28→    maximumFractionDigits: 2,
    29→  }).format(v);
    30→};
    31→
    32→const fmtNumber = (n: number, decimals = 2): string => {
    33→  const v = Number.isFinite(n) ? n : 0;
    34→  return new Intl.NumberFormat("es-CO", {
    35→    minimumFractionDigits: decimals,
    36→    maximumFractionDigits: decimals,
    37→  }).format(v);
    38→};
    39→
    40→const fmtFecha = (d: Date | string | null | undefined): string => {
    41→  if (!d) return "—";
    42→  const date = typeof d === "string" ? new Date(d) : d;
    43→  if (isNaN(date.getTime())) return "—";
    44→  return new Intl.DateTimeFormat("es-CO", {
    45→    year: "numeric",
    46→    month: "2-digit",
    47→    day: "2-digit",
    48→  }).format(date);
    49→};
    50→
    51→const fmtFechaHora = (d: Date | string | null | undefined): string => {
    52→  if (!d) return "—";
    53→  const date = typeof d === "string" ? new Date(d) : d;
    54→  if (isNaN(date.getTime())) return "—";
    55→  return new Intl.DateTimeFormat("es-CO", {
    56→    year: "numeric",
    57→    month: "2-digit",
    58→    day: "2-digit",
    59→    hour: "2-digit",
    60→    minute: "2-digit",
    61→  }).format(date);
    62→};

   390→interface Col {
   391→  label: string;
   392→  width: string;
   393→  align?: "left" | "right" | "center";
   394→  render: (row: any, i: number) => string;
   395→}
   396→
   397→function TablePDF({ columns, rows }: { columns: Col[]; rows: any[] }) {
   398→  if (rows.length === 0) {
   399→    return (
   400→      <View style={[styles.table, { padding: 8 }]}>
   401→        <Text style={{ fontSize: 8, color: COLORS.textMuted }}>
   402→          Sin movimientos para mostrar.
   403→        </Text>
   404→      </View>
   405→    );
   406→  }
   407→  return (
   408→    <View style={styles.table}>
   409→      <View style={styles.tableRow}>
   410→        {columns.map((c, i) => (
   411→          <Text
   412→            key={i}
   413→            style={[
   414→              styles.tableColHeader,
   415→              { width: c.width, textAlign: c.align ?? "left" },
   416→            ]}
   417→          >
   418→            {c.label}
   419→          </Text>
   420→        ))}
   421→      </View>
   422→      {rows.map((row, ri) => (
   423→        <View
   424→          key={ri}
   425→          style={[styles.tableRow, ri % 2 === 1 ? styles.tableRowAlt : {}]}
   426→        >
   427→          {columns.map((c, ci) => (
   428→            <Text
   429→              key={ci}
   430→              style={[
   431→                styles.tableCol,
   432→                { width: c.width, textAlign: c.align ?? "left" },
   433→              ]}
   434→            >
   435→              {c.render(row, ri)}
   436→            </Text>
   437→          ))}
   438→        </View>
   439→      ))}
   440→    </View>
   441→  );
   442→}

   515→          <View style={[styles.boxCol, { flex: 2 }]}>
   516→            <Text style={styles.boxLabel}>Cliente</Text>
   517→            <Text style={styles.boxValue}>{factura.nombreCliente}</Text>

   528→                Tel: {factura.tercero.telefono}

   554→            columns={[

   623→            <TotalRow
   624→              label="Saldo pendiente"
   625→              value={fmtCOP(factura.saldoPendiente ?? 0)}
   626→            />
   627→          )}

   890→// 4. BALANCE DE PRUEBA
   891→// ============================================================
   892→
   893→export async function generarPDFBalancePrueba(
   894→  empresaId: string,
   895→): Promise<Buffer> {
   896→  const empresa = await cargarEmpresa(empresaId);
   897→  if (!empresa) throw new Error("Empresa no encontrada");
   898→
   899→  const cuentas = await db.cuentaPuc.findMany({
   900→    where: { empresaId },
   901→    orderBy: { codigo: "asc" },
   902→  });
   903→  const movs = await db.movimientoContable.findMany({
   904→    where: {
   905→      empresaId,
   906→      comprobante: { estado: "contabilizado" },
   907→    },
   908→    select: { cuentaId: true, debito: true, credito: true },
   909→  });
   910→  const sumas: Record<string, { debito: number; credito: number }> = {};
   911→  for (const m of movs) {
   912→    const cur = sumas[m.cuentaId] ?? { debito: 0, credito: 0 };
   913→    cur.debito += m.debito;
   914→    cur.credito += m.credito;
   915→    sumas[m.cuentaId] = cur;
   916→  }
   917→  const rows = cuentas.map((c) => {
   918→    const s = sumas[c.id] ?? { debito: 0, credito: 0 };
   919→    const saldo = round(
   920→      c.naturaleza === "debito" ? s.debito - s.credito : s.credito - s.debito,
   921→      2,
   922→    );
   923→    return {
   924→      ...c,
   925→      debito: round(s.debito, 2),
   926→      credito: round(s.credito, 2),
   927→      saldo,
   928→    };
   929→  });
   930→  const filtradas = rows.filter(
   931→    (r) => r.debito !== 0 || r.credito !== 0 || r.nivel <= 2,
   932→  );
   933→  const totalDebito = round(
   934→    filtradas.reduce((a, b) => a + b.debito, 0),
   935→    2,
   936→  );
   937→  const totalCredito = round(
   938→    filtradas.reduce((a, b) => a + b.credito, 0),
   939→    2,
   940→  );
   941→  const diferencia = round(totalDebito - totalCredito, 2);
   942→
   943→  const doc = (
   944→    <Document title="Balance de Prueba" author={empresa.razonSocial}>
   945→      <Page size="LETTER" style={styles.page}>
   946→        <Header
   947→          empresa={empresa}
   948→          titulo={"Balance de Prueba"}
   949→          subtitulo={`Corte a ${fmtFecha(new Date())} · ${filtradas.length} cuentas`}
   950→        />
   951→        <View style={styles.section}>
   952→          <TablePDF
   953→            columns={[
   954→              {
   955→                label: "Código",
   956→                width: "12%",
   957→                render: (r) => r.codigo,
   958→              },
   959→              {
   960→                label: "Nombre",
   961→                width: "42%",
   962→                render: (r) =>
   963→                  `${"  ".repeat(Math.max(0, (r.nivel ?? 1) - 1))}${r.nombre}`,
   964→              },
   965→              {
   966→                label: "Naturaleza",
   967→                width: "11%",
   968→                align: "center",
   969→                render: (r) => r.naturaleza,
   970→              },
   971→              {
   972→                label: "Débitos",
   973→                width: "12%",
   974→                align: "right",
   975→                render: (r) => fmtNumber(r.debito, 2),
   976→              },
   977→              {
   978→                label: "Créditos",
   979→                width: "12%",
   980→                align: "right",
   981→                render: (r) => fmtNumber(r.credito, 2),
   982→              },
   983→              {
   984→                label: "Saldo",
   985→                width: "11%",
   986→                align: "right",
   987→                render: (r) => fmtNumber(r.saldo, 2),
   988→              },
   989→            ]}
   990→            rows={filtradas}
   991→          />
   992→        </View>
   993→        <View style={[styles.totalsTable, { width: "100%", marginTop: 12 }]}>
   994→          <TotalRow
   995→            label="Total débitos"
   996→            value={fmtNumber(totalDebito, 2)}
   997→            bold
   998→          />
   999→          <TotalRow
  1000→            label="Total créditos"
  1001→            value={fmtNumber(totalCredito, 2)}
  1002→            bold
  1003→          />
  1004→          <TotalRow
  1005→            label="Diferencia"
  1006→            value={fmtNumber(diferencia, 2)}
  1007→            bold
  1008→          />
  1009→        </View>
  1010→        {Math.abs(diferencia) > 1 && (
  1011→          <View style={[styles.stampBox, { borderColor: COLORS.danger }]}>
  1012→            <Text style={styles.stampText}>BALANCE DESCUADRADO</Text>
  1013→          </View>
  1014→        )}
  1015→        <Footer />
  1016→      </Page>
  1017→    </Document>
  1018→  );
  1019→  return await renderToBuffer(doc);
  1020→}

  1713→  const ventasMes = round(facturasMes.reduce((a, f) => a + f.total, 0), 2);
  1714→  const ventasMesPasado = round(facturasMesPasado.reduce((a, f) => a + f.total, 0), 2);
  1715→  const variacion = ventasMesPasado > 0 ? round(((ventasMes - ventasMesPasado) / ventasMesPasado) * 100, 2) : 0;
  1716→  const numFacturas = facturasMes.length;
  1717→  const ticketPromedio = numFacturas > 0 ? round(ventasMes / numFacturas, 2) : 0;
  1718→
  1719→  // Cartera
  1720→  const cartera = await db.facturaVenta.findMany({
  1721→    where: { empresaId, estado: "vigente", saldoPendiente: { gt: 0 } },
  1722→    select: { saldoPendiente: true, fechaVencimiento: true },
  1723→  });
  1724→  const carteraTotal = round(cartera.reduce((a, c) => a + c.saldoPendiente, 0), 2);
  1725→  let cartera30 = 0, cartera60 = 0, cartera90 = 0, carteraMas90 = 0, carteraVencida = 0;
  1726→  for (const c of cartera) {
  1727→    if (!c.fechaVencimiento) continue;
  1728→    const dias = Math.floor((now.getTime() - c.fechaVencimiento.getTime()) / (1000 * 60 * 60 * 24));
  1729→    if (dias <= 0) continue;
  1730→    if (dias <= 30) cartera30 += c.saldoPendiente;
  1731→    else if (dias <= 60) cartera60 += c.saldoPendiente;
  1732→    else if (dias <= 90) cartera90 += c.saldoPendiente;
  1733→    else carteraMas90 += c.saldoPendiente;
  1734→    carteraVencida += c.saldoPendiente;
  1735→  }
  1736→
  1737→  // Top productos por valor vendido en el mes
  1738→  const detallesMes = await db.detalleFacturaVenta.findMany({
  1739→    where: {
  1740→      factura: {
  1741→        empresaId,
  1742→        fecha: { gte: inicioMes },
  1743→        estado: "vigente",
  1744→      },
  1745→    },
  1746→    select: {
  1747→      productoId: true,
  1748→      descripcion: true,
  1749→      cantidad: true,
  1750→      total: true,
  1751→    },
  1752→  });
  1753→  const porProducto: Record<string, { descripcion: string; cantidad: number; total: number }> = {};
  1754→  for (const d of detallesMes) {
  1755→    if (!d.productoId) continue;
  1756→    if (!porProducto[d.productoId]) {
  1757→      porProducto[d.productoId] = { descripcion: d.descripcion, cantidad: 0, total: 0 };
  1758→    }
  1759→    porProducto[d.productoId].cantidad += d.cantidad;
  1760→    porProducto[d.productoId].total += d.total;
  1761→  }
  1762→  const topProductos = Object.entries(porProducto)
  1763→    .map(([id, v]) => ({ id, ...v, total: round(v.total, 2) }))
  1764→    .sort((a, b) => b.total - a.total)
  1765→    .slice(0, 10);
  1766→
  1767→  // Compras
  1768→  const comprasMes = await db.facturaCompra.aggregate({
  1769→    where: { empresaId, fecha: { gte: inicioMes }, estado: "vigente" },
  1770→    _sum: { total: true },
  1771→  });
  1772→
  1773→  // Totales catálogo
  1774→  const totalProductos = await db.producto.count({ where: { empresaId, activo: true } });
  1775→  const totalClientes = await db.tercero.count({ where: { empresaId, tipo: "cliente", activo: true } });
  1776→  const totalProveedores = await db.tercero.count({ where: { empresaId, tipo: "proveedor", activo: true } });
  1777→
  1778→  // Stock crítico
  1779→  const stockCritico = await db.producto.findMany({
  1780→    where: { empresaId, activo: true },
  1781→    select: {
  1782→      id: true,
  1783→      codigo: true,
  1784→      nombre: true,
  1785→      stockMinimo: true,
  1786→      saldosInventario: { select: { cantidad: true } },
  1787→    },
  1788→  });
  1789→  const criticos = stockCritico
  1790→    .map((p) => ({
  1791→      codigo: p.codigo,
  1792→      nombre: p.nombre,
  1793→      stockTotal: p.saldosInventario.reduce((a, s) => a + s.cantidad, 0),
  1794→      stockMinimo: p.stockMinimo,
  1795→    }))
  1796→    .filter((p) => p.stockTotal <= p.stockMinimo)
  1797→    .slice(0, 10);
  1798→
  1799→  // KPIs contables del mes
  1800→  const contaMovs = await db.movimientoContable.aggregate({
  1801→    where: { empresaId, fecha: { gte: inicioMes }, comprobante: { estado: "contabilizado" } },
  1802→    _sum: { debito: true, credito: true },
  1803→  });
  1804→
  1805→  const doc = (
  1806→    <Document title="Reporte Ejecutivo" author={empresa.razonSocial}>
  1807→      <Page size="LETTER" style={styles.page}>
  1808→        <Header
  1809→          empresa={empresa}
  1810→          titulo={"Reporte Ejecutivo"}
  1811→          subtitulo={`Período ${fmtFecha(inicioMes)} → ${fmtFecha(now)}`}
  1812→        />
  1813→
  1814→        {/* KPIs */}
  1815→        <View style={styles.section}>
  1816→          <Text style={styles.sectionTitle}>Indicadores clave del mes</Text>
  1817→          <View style={styles.boxRow}>
  1818→            <View style={styles.kpiBox}>
  1819→              <Text style={styles.kpiLabel}>Ventas del mes</Text>
  1820→              <Text style={styles.kpiValue}>{fmtCOP(ventasMes)}</Text>
  1821→              <Text style={{ fontSize: 7, color: variacion >= 0 ? COLORS.primary : COLORS.danger }}>
  1822→                {variacion >= 0 ? "▲" : "▼"} {Math.abs(variacion)}% vs mes anterior
  1823→              </Text>
  1824→            </View>
  1825→            <View style={styles.kpiBox}>
  1826→              <Text style={styles.kpiLabel}>Cartera total</Text>
  1827→              <Text style={styles.kpiValue}>{fmtCOP(carteraTotal)}</Text>
  1828→              <Text style={{ fontSize: 7, color: COLORS.textMuted }}>
  1829→                Vencida: {fmtCOP(carteraVencida)}
  1830→              </Text>
  1831→            </View>
  1832→            <View style={styles.kpiBox}>
  1833→              <Text style={styles.kpiLabel}>Compras del mes</Text>
  1834→              <Text style={styles.kpiValue}>{fmtCOP(comprasMes._sum.total ?? 0)}</Text>
  1835→              <Text style={{ fontSize: 7, color: COLORS.textMuted }}>
  1836→                {numFacturas} facturas emitidas
  1837→              </Text>
  1838→            </View>
  1839→          </View>
  1840→          <View style={styles.boxRow}>
  1841→            <View style={styles.kpiBox}>
  1842→              <Text style={styles.kpiLabel}>Ticket promedio</Text>
  1843→              <Text style={styles.kpiValue}>{fmtCOP(ticketPromedio)}</Text>
  1844→            </View>
  1845→            <View style={styles.kpiBox}>
  1846→              <Text style={styles.kpiLabel}>Productos críticos</Text>
  1847→              <Text style={styles.kpiValue}>{criticos.length}</Text>
  1848→            </View>
  1849→            <View style={styles.kpiBox}>
  1850→              <Text style={styles.kpiLabel}>Movimientos contables</Text>
  1851→              <Text style={styles.kpiValue}>{fmtCOP(contaMovs._sum.debito ?? 0)}</Text>

  1853→            </View>
  1854→          </View>
  1855→        </View>
  1856→
  1857→        {/* Cartera por edades */}
  1858→        <View style={styles.section}>
  1859→          <Text style={styles.sectionTitle}>Cartera por edades</Text>
  1860→          <View style={styles.boxRow}>
  1861→            <View style={styles.kpiBox}>
  1862→              <Text style={styles.kpiLabel}>0-30 días</Text>
  1863→              <Text style={styles.kpiValue}>{fmtCOP(cartera30)}</Text>
  1864→            </View>
  1865→            <View style={styles.kpiBox}>
  1866→              <Text style={styles.kpiLabel}>31-60 días</Text>
  1867→              <Text style={styles.kpiValue}>{fmtCOP(cartera60)}</Text>
  1868→            </View>
  1869→            <View style={styles.kpiBox}>
  1870→              <Text style={styles.kpiLabel}>61-90 días</Text>
  1871→              <Text style={styles.kpiValue}>{fmtCOP(cartera90)}</Text>
  1872→            </View>
  1873→            <View style={styles.kpiBox}>
  1874→              <Text style={styles.kpiLabel}>+90 días</Text>
  1875→              <Text style={[styles.kpiValue, { color: COLORS.danger }]}>
  1876→                {fmtCOP(carteraMas90)}
  1877→              </Text>
  1878→            </View>
  1879→          </View>
  1880→        </View>
  1881→
  1882→        {/* Top productos */}
  1883→        <View style={styles.section}>

  1912→        {/* Resumen */}

  1922→          <View style={styles.boxCol}>
  1923→            <Text style={styles.boxLabel}>Productos activos</Text>

  1928→        <Text style={styles.notaText}>
