    77→    77→// ---------- Carga de certificado PFX/P12 ----------
    78→    78→
    79→    79→/**
    80→    80→ * Carga un certificado PFX/P12 (base64 o Buffer) usando `node-forge`
    81→    81→ * para parsear la estructura PKCS#12, extraer la clave privada y
    82→    82→ * el X.509, y devolverlos en formato PEM. Devuelve la info necesaria
    83→    83→ * para firmar.
    84→    84→ *
    85→    85→ * Si la password es incorrecta, lanza "PKCS#12 password invalid".
    86→    86→ */
    87→    87→export function loadCertificate(
    88→    88→  p12Base64: string | Buffer,
    89→    89→  password: string,
    90→    90→): CertificateInfo {
    91→    91→  const pfxBuffer = Buffer.isBuffer(p12Base64)
    92→    92→    ? p12Base64
    93→    93→    : Buffer.from(p12Base64, "base64");
    94→    94→
    95→    95→  // node-forge trabaja con strings binarios. Convertimos el buffer.
    96→    96→  const p12Der = pfxBuffer.toString("binary");
    97→    97→  const asn1 = forge.asn1.fromDer(p12Der);
    98→    98→  let pfx: any;
    99→    99→  try {
   100→   100→    pfx = forge.pkcs12.pkcs12FromAsn1(asn1, password);
   101→   101→  } catch (e: any) {
   102→   102→    throw new Error(
   103→   103→      "No se pudo abrir el P12. Verifica la contraseña. (" +
   104→   104→        (e?.message ?? "unknown") +
   105→   105→        ")",
   106→   106→    );
   107→   107→  }
   108→   108→
   109→   109→  // Recorrer safeContents -> safeBags buscando certificado y private key.
   110→   110→  let forgeCert: any = null;
   111→   111→  let forgeKey: any = null;
   112→   112→  for (const sc of pfx.safeContents ?? []) {
   113→   113→    for (const bag of sc.safeBags ?? []) {
   114→   114→      if (bag.cert && !forgeCert) forgeCert = bag.cert;
   115→   115→      if (bag.key && !forgeKey) forgeKey = bag.key;
   116→   116→    }
   117→   117→  }
   118→   118→  if (!forgeCert) {
   119→   119→    throw new Error(
   120→   120→      "El P12 no contiene un certificado X.509 en un bag accesible.",
   121→   121→    );
   122→   122→  }
   123→   123→  if (!forgeKey) {
   124→   124→    throw new Error(
   125→   125→      "El P12 no contiene una clave privada accesible. ¿Está protegido con encryptionSafeContainers que no soportamos?",
   126→   126→    );
   127→   127→  }
   128→   128→
   129→   129→  const certPem = forge.pki.certificateToPem(forgeCert);
   130→   130→  const keyPem = forge.pki.privateKeyToPem(forgeKey);
   131→   131→  // DER del certificado (para fingerprint)
   132→   132→  const certDer = forge.asn1.toDer(
   133→   133→    forge.pki.certificateToAsn1(forgeCert),
   134→   134→  ).getBytes();
   135→   135→  const certDerBuf = Buffer.from(certDer, "binary");
   136→   136→
   137→   137→  // Para subject/issuer/serial: usamos crypto.X509Certificate que
   138→   138→  // parsea ASN.1 estándar y devuelve DN legible.
   139→   139→  const x509 = new X509Certificate(certDerBuf);
   140→   140→  const fingerprint = createHash("sha256").update(certDerBuf).digest("hex");
   141→   141→
   142→   142→  return {
   143→   143→    certificatePem: certPem.replace(/\r/g, "").trim() + "\n",
   144→   144→    privateKeyPem: keyPem,
   145→   145→    subject: x509.subject,
   146→   146→    issuer: x509.issuer,
   147→   147→    serialNumber: x509.serialNumber,
   148→   148→    fingerprintSha256: fingerprint,
   149→   149→    validFrom: new Date(x509.validFrom),
   150→   150→    validTo: new Date(x509.validTo),
   151→   151→  };
   152→   152→}
   153→   153→
   154→   154→// ---------- Firma XAdES-EPES envolvente sobre <Invoice> ----------
   155→   155→
   156→   156→/**
   157→   157→ * Aplica firma XAdES-EPES envolvente al elemento raíz <Invoice>
   158→   158→ * con los SignedProperties requeridos por la DIAN Colombia:
   159→   159→ *   - signingTime (fecha de firma ISO)
   160→   160→ *   - signingCertificate (digest SHA-256 del X509)
   161→   161→ *   - dataObjectFormat (descripción del documento)
   162→   162→ *
   163→   163→ * El resultado es el XML original con <ds:Signature> insertado como
   164→   164→ * hijo de <Invoice>, y una referencia con URI vacía al documento entero.
   165→   165→ */
   166→   166→export function signXMLWithXAdES(
   167→   167→  xml: string,
   168→   168→  cert: CertificateInfo,
   169→   169→  cufe: string,
   170→   170→): string {
   171→   171→  const signingTime = new Date().toISOString();
   172→   172→  const certDigestSha256 = createHash("sha256")
   173→   173→    .update(stripPemHeaders(cert.certificatePem))
   174→   174→    .digest("base64");
   175→   175→
   176→   176→  // XAdES Qualifying Properties (envueltas en ds:Object)
   177→   177→  const qualifyingProperties = `\
   178→   178→<xades:QualifyingProperties Target="#xmldsig-${cufe}" xmlns:xades="http://uri.etsi.org/01903/v1.3.2#" xmlns:xades141="http://uri.etsi.org/01903/v1.4.1#">
   179→   179→  <xades:SignedProperties Id="xmldsig-${cufe}-signedprops">
   180→   180→    <xades:SignedSignatureProperties>
   181→   181→      <xades:SigningTime>${signingTime}</xades:SigningTime>
   182→   182→      <xades:SigningCertificate>
   183→   183→        <xades:Cert>
   184→   184→          <xades:CertDigest>
   185→   185→            <ds:DigestMethod Algorithm="http://www.w3.org/2001/04/xmlenc#sha256"/>
   186→   186→            <ds:DigestValue>${certDigestSha256}</ds:DigestValue>
   187→   187→          </xades:CertDigest>
   188→   188→          <xades:IssuerSerial>
   189→   189→            <ds:X509IssuerName>${escapeXml(cert.issuer)}</ds:X509IssuerName>
   190→   190→            <ds:X509SerialNumber>${escapeXml(cert.serialNumber)}</ds:X509SerialNumber>
   191→   191→          </xades:IssuerSerial>
   192→   192→        </xades:Cert>
   193→   193→      </xades:SigningCertificate>
   194→   194→      <xades:SignaturePolicyIdentifier>
   195→   195→        <xades:SignaturePolicyImplied/>
   196→   196→      </xades:SignaturePolicyIdentifier>
   197→   197→    </xades:SignedSignatureProperties>
   198→   198→    <xades:SignedDataObjectProperties>
   199→   199→      <xades:DataObjectFormat ObjectReference="#ref-${cufe}">
   200→   200→        <xades:MimeType>text/xml</xades:MimeType>
   201→   201→        <xades:Encoding>UTF-8</xades:Encoding>
   202→   202→        <xades:Description>Factura Electrónica de Venta UBL 2.1</xades:Description>
   203→   203→      </xades:DataObjectFormat>
   204→   204→    </xades:SignedDataObjectProperties>
   205→   205→  </xades:SignedProperties>
   206→   206→</xades:QualifyingProperties>`;
   207→   207→
   208→   208→  const sig = new SignedXml({
   209→   209→    privateKey: cert.privateKeyPem,
   210→   210→    publicCert: cert.certificatePem,
   211→   211→    signatureAlgorithm: "http://www.w3.org/2001/04/xmldsig-more#rsa-sha256",
   212→   212→    canonicalizationAlgorithm: "http://www.w3.org/TR/2001/REC-xml-c14n-20010315",
   213→   213→    // XAdES Qualifying Properties van dentro de ds:Object
   214→   214→    objects: [
   215→   215→      {
   216→   216→        content: qualifyingProperties,
   217→   217→        attributes: { Id: `xmldsig-${cufe}-object` },
   218→   218→      },
   219→   219→    ],
   220→   220→  });
   221→   221→
   222→   222→  // Referencia 1: documento firmado (URI vacía = documento completo)
   223→   223→  sig.addReference({
   224→   224→    xpath: "//*[local-name()='Invoice']",
   225→   225→    transforms: ["http://www.w3.org/2000/09/xmldsig#enveloped-signature", "http://www.w3.org/TR/2001/REC-xml-c14n-20010315"],
   226→   226→    digestAlgorithm: "http://www.w3.org/2001/04/xmlenc#sha256",
   227→   227→    uri: "",
   228→   228→    isEmptyUri: true,
   229→   229→    id: `ref-${cufe}`,
   230→   230→  });
   231→   231→
   232→   232→  // Referencia 2: SignedProperties (XAdES)
   233→   233→  sig.addReference({
   234→   234→    xpath: "//*[local-name()='SignedProperties']",
   235→   235→    transforms: ["http://www.w3.org/TR/2001/REC-xml-c14n-20010315"],
   236→   236→    digestAlgorithm: "http://www.w3.org/2001/04/xmlenc#sha256",
   237→   237→    uri: `#xmldsig-${cufe}-signedprops`,
   238→   238→    isEmptyUri: false,
   239→   239→    type: "http://uri.etsi.org/01903#SignedProperties",
   240→   240→  });
   241→   241→
   242→   242→  sig.computeSignature(xml, {
   243→   243→    prefix: "ds",
   244→   244→    attrs: { Id: `xmldsig-${cufe}` },
   245→   245→    location: {
   246→   246→      reference: "//*[local-name()='Invoice']",
   247→   247→      action: "prepend",
   248→   248→    },
   249→   249→    existingPrefixes: {
   250→   250→      ds: "http://www.w3.org/2000/09/xmldsig#",
   251→   251→      xades: "http://uri.etsi.org/01903/v1.3.2#",
   252→   252→    },
   253→   253→  });
   254→   254→
   255→   255→  return sig.getSignedXml();
   256→   256→}

   384→   384→// ---------- POST a DIAN con manejo de errores ----------
   385→   385→
   386→   386→async function postToDian(
   387→   387→  endpoint: string,
   388→   388→  soapEnvelope: string,
   389→   389→  soapAction: string,
   390→   390→  expectedResponseLocalName: string,
   391→   391→): Promise<DIANResponse> {
   392→   392→  try {
   393→   393→    const controller = new AbortController();
   394→   394→    const timeout = setTimeout(() => controller.abort(), 60_000);
   395→   395→
   396→   396→    const response = await fetch(endpoint, {
   397→   397→      method: "POST",
   398→   398→      headers: {
   399→   399→        "Content-Type": `application/soap+xml; charset=utf-8; action="${soapAction}"`,
   400→   400→        "Accept-Encoding": "gzip,deflate",
   401→   401→        "Content-Length": String(Buffer.byteLength(soapEnvelope, "utf8")),
   402→   402→      },
   403→   403→      body: soapEnvelope,
   404→   404→      signal: controller.signal,
   405→   405→    });
   406→   406→
   407→   407→    clearTimeout(timeout);
   408→   408→
   409→   409→    const text = await response.text();
   410→   410→    return parseDianResponse(text, expectedResponseLocalName);
   411→   411→  } catch (e: any) {
   412→   412→    return {
   413→   413→      isAccepted: false,
   414→   414→      errorMessage:
   415→   415→        "Error de red/timeout contactando DIAN: " + (e?.message ?? String(e)),
   416→   416→    };
   417→   417→  }
   418→   418→}
   419→   419→
   420→   420→/**
   421→   421→ * Extrae trackId, status, isValid y errores del XML de respuesta SOAP.
   422→   422→ * La DIAN responde con un objeto que tiene `StatusCode`, `StatusDescription`,
   423→   423→ * y un `ErrorMessage`-array con `statusCode`/`description`/`correspondence`-id.
   424→   424→ */
   425→   425→function parseDianResponse(
   426→   426→  responseXml: string,
   427→   427→  expectedLocalName: string,
   428→   428→): DIANResponse {
   429→   429→  const clean = responseXml.replace(/<\?xml[^>]*\?>/, "").trim();
   430→   430→  // Quick regex-based extraction (avoid xpath dependency).
   431→   431→  const pick = (tag: string) => {
   432→   432→    const re = new RegExp(
   433→   433→      `<[^:>]*:?${tag}[^>]*>([^<]*)</[^:>]*:?${tag}>`,
   434→   434→      "i",
   435→   435→    );
   436→   436→    const m = clean.match(re);
   437→   437→    return m ? m[1].trim() : undefined;
   438→   438→  };
   439→   439→
   440→   440→  const trackId = pick("TrackId");
   441→   441→  const statusCode = pick("StatusCode");
   442→   442→  const statusMessage = pick("StatusMessage") ?? pick("StatusDescription");
   443→   443→  const cufe = pick("UUID");
   444→   444→  const errorMessageDesc = pick("Description");
   445→   445→
   446→   446→  // 00..99 -> DIAN acceptance: 00..09 = aceptado; 99, 100.. = error
   447→   447→  // Convención estándar DIAN: isValid == "true" si StatusCode está entre "00".."09".
   448→   448→  const accepted =
   449→   449→    statusCode !== undefined &&
   450→   450→    /^0[0-9]$/i.test(statusCode) &&
   451→   451→    statusMessage?.toLowerCase().includes("valid");
   452→   452→
   453→   453→  return {
   454→   454→    trackId,
   455→   455→    status: statusCode,
   456→   456→    cufe,
   457→   457→    errorMessage: errorMessageDesc,
   458→   458→    raw: clean,
   459→   459→    isAccepted: Boolean(accepted),
   460→   460→  };
   461→   461→}

   463→   463→// ---------- Carga de configuración DIAN desde BD ----------
   464→   464→
   465→   465→/**
   466→   466→ * Carga la configuración DIAN activa de una empresa, descifra el
   467→   467→ * certificado P12 y la contraseña, e instancia el CertificateInfo.
   468→   468→ * Devuelve null si no hay configuración activa o no hay certificado.
   469→   469→ */
   470→   470→export async function loadDIANConfigFromDB(
   471→   471→  empresaId: string,
   472→   472→): Promise<{ config: DIANConfig; configRecord: any } | null> {
   473→   473→  const record = await db.configuracionDIAN.findFirst({
   474→   474→    where: {
   475→   475→      empresaId,
   476→   476→      activo: true,
   477→   477→    },
   478→   478→    orderBy: { updatedAt: "desc" },
   479→   479→  });
   480→   480→  if (!record || !record.certificadoP12Enc || !record.certificadoPasswordEnc) {
   481→   481→    return null;
   482→   482→  }
   483→   483→

   485→   485→    const p12B64 = decrypt(record.certificadoP12Enc);

   487→   487→    const cert = loadCertificate(p12B64, password);
   488→   488→
   489→   489→    return {
   490→   490→      config: {
   491→   491→        ambiente: record.ambiente as "pruebas" | "produccion",

   493→   493→        proveedorTecnologicoToken: record.proveedorTecnologicoTokenEnc
   494→   494→          ? decrypt(record.proveedorTecnologicoTokenEnc)
   495→   495→          : undefined,
   496→   496→        softwareId: record.softwareId ?? undefined,
   497→   497→        softwarePin: record.softwarePin ?? undefined,
   498→   498→        webhookUrl: record.webhookUrl ?? undefined,
   499→   499→        certificate: cert,
   500→   500→      },
   501→   501→      configRecord: record,
   502→   502→    };
   503→   503→  } catch (e: any) {
   504→   504→    console.error(
   505→   505→      "[dian-real] No se pudo cargar el certificado:",
   506→   506→      e?.message ?? e,
   507→   507→    );
   508→   508→    return null;
   509→   509→  }
   510→   510→}

   514→   514→/**
   515→   515→ * Toma una factura ya persistida (con xmlUBL), firma con XAdES-EPES,
   516→   516→ * envía a DIAN (SendBillSync), registra en DIANEnvio y actualiza
   517→   517→ * FacturaVenta.estadoDIAN. Si falla, programa reintento.

   519→   519→export async function enviarFacturaADian(
   520→   520→  facturaId: string,
   521→   521→  empresaId: string,
   522→   522→): Promise<{ ok: boolean; envioId?: string; trackId?: string; mensaje?: string }> {
   523→   523→  const factura = await db.facturaVenta.findFirst({
   524→   524→    where: { id: facturaId, empresaId },
   525→   525→  });
   526→   526→  if (!factura) {
   527→   527→    return { ok: false, mensaje: "Factura no encontrada" };
   528→   528→  }
   529→   529→  if (!factura.xmlUBL || !factura.cufe) {
   530→   530→    return { ok: false, mensaje: "La factura no tiene XML UBL ni CUFE generado" };
   531→   531→  }
   532→   532→  if (!["borrador", "procesado", "rechazado"].includes(factura.estadoDIAN)) {
   533→   533→    // ya aceptada o enviada-aceptada, no reenviar salvo explícito
   534→   534→  }
   535→   535→
   536→   536→  const loaded = await loadDIANConfigFromDB(empresaId);
   537→   537→  if (!loaded) {
   538→   538→    return {
   539→   539→      ok: false,
   540→   540→      mensaje:
   541→   541→        "No hay configuración DIAN activa con certificado cargado. Sube tu .p12 en la pestaña Configuración.",
   542→   542→    };
   543→   543→  }
   544→   544→
   545→   545→  const { config, configRecord } = loaded;
   546→   546→
   547→   547→  // 1. Crear registro de envío (pendiente)
   548→   548→  const envio = await db.dIANEnvio.create({
   549→   549→    data: {
   550→   550→      empresaId,
   551→   551→      facturaId: factura.id,
   552→   552→      tipoDocumento: "factura_venta",
   553→   553→      cufe: factura.cufe!,
   554→   554→      estado: "enviando",
   555→   555→      intentos: 1,
   556→   556→      maxIntentos: 5,
   557→   557→      proximoIntento: null,
   558→   558→    },
   559→   559→  });

   561→   561→  try {
   562→   562→    // 2. Firmar XML
   563→   563→    const xmlFirmado = signXMLWithXAdES(factura.xmlUBL, config.certificate!, factura.cufe!);
   564→   564→    await db.dIANEnvio.update({
   565→   565→      where: { id: envio.id },
   566→   566→      data: { xmlFirmado },
   567→   567→    });
   568→   568→
   569→   569→    // 3. Enviar a DIAN
   570→   570→    const dianResp = await sendBillSyncToDIAN(xmlFirmado, config, `face_${factura.numero}.xml`);
   571→   571→
   572→   572→    if (dianResp.trackId) {
   573→   573→      await db.dIANEnvio.update({
   574→   574→        where: { id: envio.id },
   575→   575→        data: {
   576→   576→          trackId: dianResp.trackId,
   577→   577→          estado: dianResp.isAccepted ? "aceptado" : "pendiente",
   578→   578→          respuestaDian: JSON.stringify({
   579→   579→            statusCode: dianResp.status,
   580→   580→            statusMessage: dianResp.errorMessage,
   581→   581→            cufe: dianResp.cufe,
   582→   582→            raw: dianResp.raw,
   583→   583→          }),
   584→   584→          proximoIntento: dianResp.isAccepted
   585→   585→            ? null
   586→   586→            : new Date(Date.now() + 5 * 60 * 1000),
   587→   587→        },
   588→   588→      });
   589→   589→
   590→   590→      await db.facturaVenta.update({
   591→   591→        where: { id: factura.id },
   592→   592→        data: {
   593→   593→          estadoDIAN: dianResp.isAccepted ? "aceptado" : "procesado",
   594→   594→        },
   595→   595→      });
   596→   596→
   597→   597→      return {
   598→   598→        ok: true,
   599→   599→        envioId: envio.id,
   600→   600→        trackId: dianResp.trackId,
   601→   601→        mensaje: dianResp.isAccepted
   602→   602→          ? "Factura aceptada por DIAN"
   603→   603→          : `Enviada, en proceso (trackId: ${dianResp.trackId})`,
   604→   604→      };
   605→   605→    } else {
   606→   606→      // Error de envío
   607→   607→      await db.dIANEnvio.update({
   608→   608→        where: { id: envio.id },
   609→   609→        data: {
   610→   610→          estado: "error",
   611→   611→          mensajeError: dianResp.errorMessage ?? "Error desconocido",
   612→   612→          respuestaDian: dianResp.raw
   613→   613→            ? JSON.stringify({ raw: dianResp.raw })
   614→   614→            : null,
   615→   615→          proximoIntento: new Date(Date.now() + 5 * 60 * 1000),
   616→   616→        },
   617→   617→      });
   618→   618→
   619→   619→      return {
   620→   620→        ok: false,
   621→   621→        envioId: envio.id,
   622→   622→        mensaje: dianResp.errorMessage ?? "Error de envío DIAN",
   623→   623→      };
   624→   624→    }
   625→   625→  } catch (e: any) {
   626→   626→    const mensaje = e?.message ?? String(e);
   627→   627→    await db.dIANEnvio.update({
   628→   628→      where: { id: envio.id },
   629→   629→      data: {
   630→   630→        estado: "error",
   631→   631→        mensajeError: mensaje,
   632→   632→        proximoIntento: new Date(Date.now() + 5 * 60 * 1000),
   633→   633→      },
   634→   634→    });
   635→   635→    return { ok: false, envioId: envio.id, mensaje };
   636→   636→  }
   637→   637→}
   638→   638→
   639→   639→// ---------- Procesador de cola de reintentos ----------
   640→   640→
   641→   641→/**
   642→   642→ * Procesa todos los DIANEnvio pendientes o con proximoIntento vencido.
   643→   643→ * Aplica backoff exponencial: intento N -> 2^N * 60s.
   644→   644→ */
   645→   645→export async function procesarColaDIAN(
   646→   646→  empresaId: string,
   647→   647→  limite = 50,
   648→   648→): Promise<QueueProcessResult> {
   649→   649→  const now = new Date();
   650→   650→  const pendientes = await db.dIANEnvio.findMany({
   651→   651→    where: {
   652→   652→      empresaId,
   653→   653→      OR: [
   654→   654→        { estado: "pendiente" },
   655→   655→        { estado: "error" },
   656→   656→        { estado: "enviando", proximoIntento: { lte: now } },
   657→   657→      ],
   658→   658→      intentos: { lt: 5 },
   659→   659→    },
   660→   660→    take: limite,
   661→   661→    orderBy: { createdAt: "asc" },
   662→   662→  });
   663→   663→
   664→   664→  const result: QueueProcessResult = {
   665→   665→    processed: 0,
   666→   666→    accepted: 0,
   667→   667→    rejected: 0,
   668→   668→    errors: 0,
   669→   669→    rescheduled: 0,
   670→   670→    details: [],
   671→   671→  };
   672→   672→
   673→   673→  for (const envio of pendientes) {
   674→   674→    if (!envio.facturaId) continue;
   675→   675→    result.processed++;
   676→   676→
   677→   677→    const updated = await db.dIANEnvio.update({
   678→   678→      where: { id: envio.id },
   679→   679→      data: {
   680→   680→        intentos: { increment: 1 },
   681→   681→        estado: "enviando",
   682→   682→      },
   683→   683→    });
   684→   684→
   685→   685→    const res = await enviarFacturaADian(envio.facturaId, empresaId);
   686→   686→
   687→   687→    if (res.ok && res.trackId) {
   688→   688→      result.accepted++;
   689→   689→      result.details.push({
   690→   690→        envioId: envio.id,
   691→   691→        facturaId: envio.facturaId ?? undefined,
   692→   692→        estado: "aceptado",
   693→   693→        mensaje: res.mensaje,
   694→   694→      });
   695→   695→    } else {
   696→   696→      // Si ya llegó a maxIntentos, marcar como rechazado definitivo
   697→   697→      if (updated.intentos >= updated.maxIntentos) {
   698→   698→        await db.dIANEnvio.update({
   699→   699→          where: { id: envio.id },
   700→   700→          data: {
   701→   701→            estado: "rechazado",
   702→   702→            mensajeError: res.mensaje ?? "Máximos reintentos alcanzados",
   703→   703→          },
   704→   704→        });
   705→   705→        result.rejected++;
   706→   706→      } else {
   707→   707→        // Reagendar con backoff exponencial (2^n * 60s)
   708→   708→        const delayMs = Math.pow(2, updated.intentos) * 60 * 1000;
   709→   709→        await db.dIANEnvio.update({
   710→   710→          where: { id: envio.id },
   711→   711→          data: {
   712→   712→            estado: "error",
   713→   713→            mensajeError: res.mensaje,
   714→   714→            proximoIntento: new Date(Date.now() + delayMs),
   715→   715→          },
   716→   716→        });
   717→   717→        result.errors++;
   718→   718→        result.rescheduled++;
   719→   719→      }
   720→   720→      result.details.push({
   721→   721→        envioId: envio.id,
   722→   722→        facturaId: envio.facturaId ?? undefined,
   723→   723→        estado: "error",
   724→   724→        mensaje: res.mensaje,
   725→   725→      });
   726→   726→    }
   727→   727→  }
   728→   728→
   729→   729→  return result;
   730→   730→}

   750→   750→// ---------- Test de conexión DIAN ----------
   751→   751→
   752→   752→/**
   753→   753→ * Prueba la conexión con DIAN en el ambiente configurado.
   754→   754→ * Sin certificado, responde gracefully.
   755→   755→ */
   756→   756→export async function testConnectionDIAN(
   757→   757→  empresaId: string,
   758→   758→): Promise<{ ok: boolean; message: string; ambiente?: string }> {
   759→   759→  const loaded = await loadDIANConfigFromDB(empresaId);
   760→   760→  if (!loaded) {
   761→   761→    return {
   762→   762→      ok: false,
   763→   763→      message:
   764→   764→        "No hay certificado cargado. Sube tu .p12 en la pestaña Configuración.",
   765→   765→    };
   766→   766→  }
   767→   767→
   768→   768→  const { config, configRecord } = loaded;
   769→   769→
   770→   770→  // Validar vigencia del certificado
   771→   771→  const now = new Date();
   772→   772→  if (config.certificate!.validFrom > now) {
   773→   773→    return {
   774→   774→      ok: false,
   775→   775→      message: `El certificado aún no es válido (vigente desde ${config.certificate!.validFrom.toISOString().slice(0, 10)})`,
   776→   776→      ambiente: config.ambiente,
   777→   777→    };
   778→   778→  }
   779→   779→  if (config.certificate!.validTo < now) {
   780→   780→    return {
   781→   781→      ok: false,
   782→   782→      message: `El certificado expiró el ${config.certificate!.validTo.toISOString().slice(0, 10)}. Renueva el P12 con tu CA.`,
   783→   783→      ambiente: config.ambiente,
   784→   784→    };
   785→   785→  }
   786→   786→
   787→   787→  // Si hay token PAC, hacemos ping real a DIAN con un trackId falso

   789→   789→  if (config.proveedorTecnologicoToken) {
   790→   790→    const endpoint = DIAN_ENDPOINTS[config.ambiente];
   791→   791→    try {
   792→   792→      const controller = new AbortController();
   793→   793→      const timeout = setTimeout(() => controller.abort(), 10_000);
   794→   794→      const resp = await fetch(endpoint, {
   795→   795→        method: "POST",
   796→   796→        headers: { "Content-Type": "text/xml; charset=utf-8" },
   797→   797→        body: '<?xml version="1.0" encoding="UTF-8"?><s:Envelope xmlns:s="http://www.w3.org/2003/05/soap-envelope"><s:Body/></s:Envelope>',
   798→   798→        signal: controller.signal,
   799→   799→      });
   800→   800→      clearTimeout(timeout);
   801→   801→      // Cualquier respuesta HTTP (incluso 400/500) indica reachability
   802→   802→      return {
   803→   803→        ok: true,
   804→   804→        message: `Conexión exitosa con DIAN ${config.ambiente}. Respuesta HTTP ${resp.status}. Certificado: ${config.certificate!.subject.slice(0, 60)}...`,
   805→   805→        ambiente: config.ambiente,
   806→   806→      };
   807→   807→    } catch (e: any) {
   808→   808→      return {
   809→   809→        ok: false,
   810→   810→        message: `No se pudo conectar con DIAN: ${e?.message ?? e}`,
   811→   811→        ambiente: config.ambiente,
   812→   812→      };
   813→   813→    }
   814→   814→  }
   815→   815→
   816→   816→  // Hay certificado pero no PAC: certificamos la firma localmente
   817→   817→  return {
   818→   818→    ok: true,
   819→   819→    message: `Certificado cargado y vigente (subject: ${config.certificate!.subject.slice(0, 60)}...). Sin token PAC: el envío real requiere credenciales de proveedor tecnológico autorizado.`,
   820→   820→    ambiente: config.ambiente,
   821→   821→  };
   822→   822→}
   823→   823→
   824→   824→// ---------- Guardar configuración desde multipart ----------
   825→   825→
   826→   826→/**
   827→   827→ * Recibe los campos del form de configuración DIAN, valida y guarda.
   828→   828→ * Cifra el P12 y la contraseña con AES-256-GCM antes de persistir.
   829→   829→ */
   830→   830→export async function saveDIANConfig(params: {
   831→   831→  empresaId: string;
   832→   832→  ambiente: "pruebas" | "produccion";
   833→   833→  certificadoP12B64?: string; // base64 del .p12
   834→   834→  certificadoPassword?: string;
   835→   835→  proveedorTecnologicoNit?: string;
   836→   836→  proveedorTecnologicoToken?: string;
   837→   837→  softwareId?: string;
   838→   838→  softwarePin?: string;
   839→   839→  webhookUrl?: string;
   840→   840→  activo?: boolean;
   841→   841→}): Promise<{ ok: boolean; config?: any; mensaje: string }> {
   842→   842→  const {
   843→   843→    empresaId,
   844→   844→    ambiente,
   845→   845→    certificadoP12B64,
   846→   846→    certificadoPassword,
   847→   847→    proveedorTecnologicoNit,
   848→   848→    proveedorTecnologicoToken,
   849→   849→    softwareId,
   850→   850→    softwarePin,
   851→   851→    webhookUrl,
   852→   852→    activo = true,
   853→   853→  } = params;
   854→   854→
   855→   855→  let certSubject: string | undefined;
   856→   856→  let certValidFrom: Date | undefined;
   857→   857→  let certValidTo: Date | undefined;
   858→   858→  let p12Enc: string | undefined;
   859→   859→  let passEnc: string | undefined;
   860→   860→
   861→   861→  if (certificadoP12B64 && certificadoPassword) {
   862→   862→    // Carga y valida el certificado antes de guardar
   863→   863→    const info = loadCertificate(certificadoP12B64, certificadoPassword);
   864→   864→    certSubject = info.subject;
   865→   865→    certValidFrom = info.validFrom;
   866→   866→    certValidTo = info.validTo;
   867→   867→    p12Enc = encrypt(certificadoP12B64);
   868→   868→    passEnc = encrypt(certificadoPassword);
   869→   869→  }
   870→   870→
   871→   871→  // Si se activa esta config, desactivar las demás de la empresa
   872→   872→  if (activo) {
   873→   873→    await db.configuracionDIAN.updateMany({
   874→   874→      where: { empresaId, activo: true },
   875→   875→      data: { activo: false },
   876→   876→    });
   877→   877→  }
   878→   878→
   879→   879→  // Upsert por (empresaId, ambiente)
   880→   880→  const existing = await db.configuracionDIAN.findUnique({
   881→   881→    where: {
   882→   882→      empresaId_ambiente: { empresaId, ambiente },
   883→   883→    },
   884→   884→  });
   885→   885→
   886→   886→  const data: any = {
   887→   887→    ambiente,
   888→   888→    proveedorTecnologicoNit,
   889→   889→    proveedorTecnologicoTokenEnc: proveedorTecnologicoToken
   890→   890→      ? encrypt(proveedorTecnologicoToken)
   891→   891→      : existing?.proveedorTecnologicoTokenEnc ?? null,
   892→   892→    softwareId,
   893→   893→    softwarePin,
   894→   894→    webhookUrl,
   895→   895→    activo,
   896→   896→  };
   897→   897→  if (p12Enc) data.certificadoP12Enc = p12Enc;
